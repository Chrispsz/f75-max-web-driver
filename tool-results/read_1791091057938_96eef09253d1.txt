    77→
    78→export default function F75App() {
    79→  const driverRef = useRef<F75Driver | null>(null);
    80→  const resetTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    81→
    82→  const [section, setSection] = useState<SectionId>("device");
    83→  const [supported, setSupported] = useState<boolean | null>(null);
    84→  const [inIframe, setInIframe] = useState(false);
    85→  const [driverReady, setDriverReady] = useState(false);
    86→  const [status, setStatus] = useState<DriverStatus>({ sim: false, wiredCommand: false, wiredDisplay: false, dongle: false });
    87→  const [busy, setBusy] = useState<string | null>(null);
    88→  const [battery, setBattery] = useState<number | null>(null);
    89→  const [routeLabel, setRouteLabel] = useState<string | null>(null);
    90→
    91→  const [rgb, setRgbState] = useState<RgbSettings>({ mode: 1, brightness: 3, speed: 2, direction: 0, colorful: false, color: hexToInt(CYAN) });
    92→  const [perf, setPerfState] = useState<PerfState>({ level: 2, sleep: 2, game: false, lockAltTab: false, lockAltF4: false, lockWin: false });
    93→  const [resetArmed, setResetArmed] = useState(false);
    94→
    95→  const setRgb = useCallback((patch: Partial<RgbSettings>) => setRgbState((prev) => ({ ...prev, ...patch })), []);
    96→  const setPerf = useCallback((patch: Partial<PerfState>) => setPerfState((prev) => ({ ...prev, ...patch })), []);
    97→
    98→  /* ------------------------------- bootstrap ------------------------------- */
    99→
   100→  useEffect(() => {
   101→    try {
   102→      const raw = localStorage.getItem(UI_KEY);
   103→      if (raw) {
   104→        const saved = JSON.parse(raw) as { section?: SectionId; rgb?: Partial<RgbSettings>; perf?: Partial<PerfState> };
   105→        if (saved.section && SECTIONS.some((s) => s.id === saved.section)) setSection(saved.section);
   106→        if (saved.rgb) setRgbState((prev) => ({ ...prev, ...saved.rgb }));
   107→        if (saved.perf) setPerfState((prev) => ({ ...prev, ...saved.perf }));
   108→      }
   109→    } catch {
   110→      /* primeira execução */
   111→    }
   112→
   113→    const driver = new F75Driver();
   114→    driverRef.current = driver;
   115→    setDriverReady(true);
   116→    driver.onStatus = (s) => setStatus(s);
   117→    driver.onBattery = (p) => setBattery(p);
   118→    setSupported(F75Driver.supported);
   119→    try {
   120→      setInIframe(window.self !== window.top);
   121→    } catch {
   122→      setInIframe(true);
   123→    }
   124→    f75log.info("F75 Max Web Driver v3 pronto — protocolo nativo portado, 100% local. Console espelhado aqui e no F12 (filtro [F75]).");
   125→    driver
   126→      .reconnectSaved(true)
   127→      .then((s) => {
   128→        if (s.wiredCommand || s.wiredDisplay || s.dongle) f75log.ok("Dispositivos já autorizados rebinados automaticamente.");
   129→      })
   130→      .catch(() => undefined);
   131→    return () => {
   132→      driver.onStatus = null;
   133→      driver.onBattery = null;
   134→    };
   135→  }, []);
   136→
   137→  useEffect(() => {
   138→    try {
   139→      localStorage.setItem(UI_KEY, JSON.stringify({ section, rgb, perf }));
   140→    } catch {
   141→      /* storage cheio/bloqueado — segue sem persistir */
   142→    }
   143→  }, [section, rgb, perf]);
   144→
   145→  const run = useCallback(
   146→    async (label: string, action: () => Promise<void>) => {
   147→      if (busy) return;
   148→      setBusy(label);
   149→      try {
   150→        await action();
   151→      } catch (err) {
   152→        const message = err instanceof F75Error ? err.message : err instanceof Error ? err.message : String(err);
   153→        f75log.err(`✗ ${label}: ${message}`);
   154→      } finally {
   155→        setBusy(null);
   156→      }
   157→    },
   158→    [busy]
   159→  );
   160→
   161→  /* --------------------------------- ações --------------------------------- */
   162→
   163→  const connect = () =>
   164→    void run("connect", async () => {
   165→      const s = await driverRef.current!.connectPicker();
   166→      if (s.dongle) await driverRef.current!.queryBattery(true).catch(() => undefined);
   167→      setRouteLabel(driverRef.current!.dongleRouteLabel);
   168→    });
   169→
   170→  const reconnect = () =>
   171→    void run("reconnect", async () => {
   172→      const s = await driverRef.current!.reconnectSaved(false);
   173→      if (s.dongle) await driverRef.current!.queryBattery(true).catch(() => undefined);
   174→      setRouteLabel(driverRef.current!.dongleRouteLabel);
   175→    });
   176→
   177→  const probeDongle = () =>
   178→    void run("probe", async () => {
   179→      await driverRef.current!.queryBattery(false);
   180→      setRouteLabel(driverRef.current!.dongleRouteLabel);
   181→    });
   182→
   183→  const disconnect = () =>
   184→    void run("disconnect", async () => {
   185→      setBattery(null);
   186→      await driverRef.current!.disconnect();
   187→    });
   188→
   189→  const toggleSim = () =>
   190→    void run("sim", async () => {
   191→      setBattery(null);
   192→      if (driverRef.current!.isSim) await driverRef.current!.disconnect();
   193→      else await driverRef.current!.enableSim();
   194→    });
   195→
   196→  const applyRgb = () => void run("rgb", () => driverRef.current!.applyRGB(rgb));
   197→
   198→  const applyPerf = () =>
   199→    void run("perf", () =>
   200→      driverRef.current!.applyPerformance({
   201→        level: perf.level,
   202→        sleep: perf.sleep,
   203→        game: perf.game,
   204→        lockAltTab: perf.lockAltTab,
   205→        lockAltF4: perf.lockAltF4,
   206→        lockWin: perf.lockWin,
   207→      })
   208→    );
   209→
   210→  const unlockAll = () => {
   211→    setPerf({ game: false, lockAltTab: false, lockAltF4: false, lockWin: false });
   212→    void run("perf", () =>
   213→      driverRef.current!.applyPerformance({ level: perf.level, sleep: perf.sleep, game: false, lockAltTab: false, lockAltF4: false, lockWin: false })
   214→    );
   215→  };
   216→
   217→  const factoryReset = () => {
   218→    if (!resetArmed) {
   219→      setResetArmed(true);
   220→      if (resetTimer.current) clearTimeout(resetTimer.current);
   221→      resetTimer.current = setTimeout(() => setResetArmed(false), 6000);
   222→      return;
   223→    }
   224→    if (resetTimer.current) clearTimeout(resetTimer.current);
   225→    setResetArmed(false);
   226→    void run("reset", async () => {
   227→      await driverRef.current!.factoryReset((stage) => f75log.info(`· ${stage}`));
   228→    });
   229→  };
   230→
   231→  const syncClock = () => void run("clock", () => driverRef.current!.syncClock());
   232→
   233→  /* --------------------------------- estado -------------------------------- */
   234→
   235→  const wiredReady = status.wiredCommand && status.wiredDisplay;
   236→  const anyConnected = status.sim || status.wiredCommand || status.wiredDisplay || status.dongle;
   237→  const wirelessReady = status.sim || status.dongle;
   238→
   239→  const connectionLabel = status.sim
   240→    ? "Simulação"
   241→    : status.dongle && wiredReady
   242→      ? "2.4G + Cabo"
   243→      : status.dongle
   244→        ? "Receiver 2.4G"
   245→        : wiredReady
   246→          ? "Cabo USB-C"
   247→          : anyConnected
   248→            ? "Parcial"
   249→            : "Desconectado";
   250→
   251→  return (
   252→    <div className="flex h-dvh flex-col overflow-hidden bg-zinc-950 text-zinc-100 selection:bg-emerald-500/30">
   253→      {/* ------------------------------- topbar ------------------------------- */}
   254→      <header className="flex h-14 shrink-0 items-center gap-3 border-b border-zinc-800/80 px-4 sm:px-5">
   255→        <span className="flex h-8 w-8 items-center justify-center rounded-lg border border-emerald-500/40 bg-emerald-500/10">
   256→          <Keyboard className="h-4 w-4 text-emerald-400" />
   257→        </span>
   258→        <div className="min-w-0">
   259→          <h1 className="truncate text-sm font-bold leading-tight">
   260→            F75 Max <span className="text-emerald-400">Web Driver</span>
   261→          </h1>
   262→          <p className="truncate text-[10px] leading-tight text-zinc-500">protocolo nativo portado · 100% local no navegador</p>
   263→        </div>
   264→
   265→        <div className="ml-auto flex items-center gap-2">
   266→          {status.dongle && !status.sim && (
   267→            <Chip ok={true} title="Bateria do teclado via receiver">
   268→              <BatteryGauge percent={battery} />
   269→            </Chip>
   270→          )}
   271→          <Chip ok={anyConnected} title="Estado da conexão">
   272→            <span className={`h-1.5 w-1.5 rounded-full ${anyConnected ? "bg-emerald-400" : "bg-zinc-600"}`} />
   273→            {connectionLabel}
   274→          </Chip>
   275→        </div>
   276→      </header>
   277→
   278→      <div className="flex min-h-0 flex-1">
   279→        {/* ------------------------------ sidebar ------------------------------ */}
   280→        <aside className="hidden w-56 shrink-0 flex-col justify-between border-r border-zinc-800/80 p-3 md:flex">
   281→          <nav aria-label="Seções do driver" className="space-y-1">
   282→            {SECTIONS.map((s) => (
   283→              <button
   284→                key={s.id}
   285→                type="button"
   286→                onClick={() => setSection(s.id)}
   287→                aria-current={section === s.id ? "page" : undefined}
   288→                className={`flex w-full items-center gap-3 rounded-lg px-3 py-2 text-left text-sm transition-colors ${
   289→                  section === s.id
   290→                    ? "bg-emerald-500/10 font-semibold text-emerald-300"
   291→                    : "text-zinc-400 hover:bg-zinc-900 hover:text-zinc-200"
   292→                }`}
   293→              >
   294→                <s.icon className="h-4 w-4 shrink-0" />
   295→                {s.label}
   296→              </button>
   297→            ))}
   298→          </nav>
   299→          <div className="space-y-2 px-1">
   300→            <div className="flex flex-wrap gap-1.5">
   301→              <Chip ok={supported === true} title={supported ? "WebHID disponível" : "Use Chrome/Chromium/Edge/Brave"}>
   302→                WebHID
   303→              </Chip>
   304→              <Chip title="Nenhuma telemetria — nada sai da máquina">100% local</Chip>
   305→            </div>
   306→            <p className="font-mono text-[10px] leading-relaxed text-zinc-600">v4 · rota do receiver validada por sonda</p>
   307→          </div>
   308→        </aside>
   309→
   310→        {/* --------------------------- conteúdo ------------------------------ */}
   311→        <div className="flex min-w-0 flex-1 flex-col">
   312→          <nav aria-label="Seções do driver" className="scrollbar-thin flex shrink-0 gap-1 overflow-x-auto border-b border-zinc-800/80 px-3 py-2 md:hidden">
   313→            {SECTIONS.map((s) => (
   314→              <button
   315→                key={s.id}
   316→                type="button"
   317→                onClick={() => setSection(s.id)}
   318→                aria-current={section === s.id ? "page" : undefined}
   319→                className={`flex shrink-0 items-center gap-1.5 rounded-full border px-3 py-1.5 text-xs transition-colors ${
   320→                  section === s.id
   321→                    ? "border-emerald-500/50 bg-emerald-500/10 font-semibold text-emerald-300"
   322→                    : "border-zinc-800 bg-zinc-900/60 text-zinc-400"
   323→                }`}
   324→              >
   325→                <s.icon className="h-3.5 w-3.5" />
   326→                {s.label}
   327→              </button>
   328→            ))}
   329→          </nav>
   330→
   331→          <main className="min-h-0 flex-1 overflow-y-auto p-4 sm:p-6">
   332→            <div className="mx-auto max-w-3xl space-y-5">
   333→              {supported === false && (
   334→                <div className="flex items-start gap-2 rounded-lg border border-amber-500/30 bg-amber-500/5 px-3 py-2.5 text-xs leading-relaxed text-amber-100/90">
   335→                  <TriangleAlert className="mt-0.5 h-4 w-4 shrink-0 text-amber-400" />
   336→                  <span>
   337→                    <strong className="text-amber-300">WebHID indisponível.</strong> Use Chrome, Chromium, Edge, Brave ou Opera — ou explore com o
   338→                    modo simulação.
   339→                  </span>
   340→                </div>
   341→              )}
   342→              {supported && inIframe && (
   343→                <div className="flex items-start gap-2 rounded-lg border border-emerald-500/25 bg-emerald-500/5 px-3 py-2.5 text-xs leading-relaxed text-emerald-100/80">
   344→                  <Cable className="mt-0.5 h-4 w-4 shrink-0 text-emerald-400" />
   345→                  <span>
   346→                    Rodando em iframe: se o seletor de dispositivos não abrir, use <strong>Open in New Tab</strong> acima do preview.
   347→                  </span>
   348→                </div>
   349→              )}
   350→
   351→              {section === "device" && (
   352→                <DevicePanel
   353→                  supported={supported}
   354→                  status={status}
   355→                  anyConnected={anyConnected}
   356→                  wiredReady={wiredReady}
   357→                  busy={busy}
   358→                  battery={battery}
   359→                  driverReady={driverReady}
   360→                  resetArmed={resetArmed}
   361→                  routeLabel={routeLabel}
   362→                  onConnect={connect}
   363→                  onReconnect={reconnect}
   364→                  onDisconnect={disconnect}
   365→                  onToggleSim={toggleSim}
   366→                  onProbe={probeDongle}
   367→                  onFactoryReset={factoryReset}
   368→                />
   369→              )}
   370→
   371→              {section === "lighting" && (
   372→                <LightingPanel rgb={rgb} setRgb={setRgb} canSend={wirelessReady} busy={busy} onApply={applyRgb} />
   373→              )}
   374→
   375→              {section === "performance" && (
   376→                <PerformancePanel perf={perf} setPerf={setPerf} canSend={wirelessReady} busy={busy} onApply={applyPerf} onUnlock={unlockAll} />
   377→              )}
   378→
   379→              {section === "display" && (
   380→                <DisplayPanel driver={driverReady ? driverRef.current : null} run={run} busy={busy} ready={status.sim || wiredReady} sim={status.sim} />
   381→              )}
   382→
   383→              {section === "keys" && <KeysPanel active={section === "keys"} />}
   384→
   385→              {section === "system" && <SystemPanel driver={driverReady ? driverRef.current : null} />}
   386→            </div>
   387→          </main>
   388→        </div>
   389→      </div>
   390→
   391→      {/* ----------------------------- status bar ----------------------------- */}
   392→      <footer className="shrink-0 border-t border-zinc-800/80 bg-zinc-950">
   393→        <div className="scrollbar-thin flex items-center gap-2 overflow-x-auto px-4 py-2 text-[11px] text-zinc-500 sm:px-5">
   394→          <Chip ok={status.wiredCommand} title="Cabo USB-C · canal de comando 0xFF13">cabo·cmd</Chip>
   395→          <Chip ok={status.wiredDisplay} title="Cabo USB-C · canal de display 0xFF68">cabo·disp</Chip>
   396→          <Chip ok={status.dongle} title="Receiver 2.4G · canais 0xFF59/0xFF60">2.4G</Chip>
   397→          {status.sim && (
   398→            <Chip ok={true} title="Modo simulação ativo">
   399→              sim
   400→            </Chip>
   401→          )}
   402→          <span className="ml-auto hidden shrink-0 items-center gap-1 font-mono text-[10px] text-zinc-600 sm:flex">
   403→            <Cable className="h-3 w-3" /> 0xFF13 · 0xFF68 · 0xFF60
   404→          </span>
   405→        </div>
   406→      </footer>
   407→    </div>
   408→  );
   409→}

   412→
   413→function DevicePanel(props: {
   414→  supported: boolean | null;
   415→  status: DriverStatus;
   416→  anyConnected: boolean;
   417→  wiredReady: boolean;
   418→  busy: string | null;
   419→  battery: number | null;
   420→  driverReady: boolean;
   421→  resetArmed: boolean;
   422→  routeLabel: string | null;
   423→  onConnect: () => void;
   424→  onReconnect: () => void;
   425→  onDisconnect: () => void;
   426→  onToggleSim: () => void;
   427→  onProbe: () => void;
   428→  onFactoryReset: () => void;
   429→}) {
   430→  const { supported, status, anyConnected, wiredReady, busy, battery, driverReady, resetArmed, routeLabel } = props;
   431→
   432→  return (
   433→    <section className="space-y-5" aria-label="Dispositivo">
   434→      <SectionHeader icon={Usb} title="Dispositivo" desc="Conexão e estado do hardware" />
   435→
   436→      <Card className="border-zinc-800 bg-zinc-900/50">
   437→        <CardContent className="space-y-4 p-4 sm:p-5">
   438→          <div className="flex flex-wrap items-center gap-2">
   439→            <Chip ok={status.wiredCommand}>cabo · comando 0xFF13</Chip>
   440→            <Chip ok={status.wiredDisplay}>cabo · display 0xFF68</Chip>
   441→            <Chip ok={status.dongle}>receiver 2.4G</Chip>
   442→            {status.dongle && !status.sim && (
   443→              <Chip ok={battery !== null}>
   444→                <BatteryGauge percent={battery} />
   445→              </Chip>
   446→            )}
   447→          </div>
   448→
   449→          {status.dongle && !status.sim && (
   450→            <div className="space-y-2 rounded-lg border border-zinc-800 bg-zinc-950/60 p-3">
   451→              <div className="flex items-center justify-between gap-2">
   452→                <p className="text-xs font-semibold text-zinc-300">Rota de comando do receiver</p>
   453→                <Button
   454→                  onClick={props.onProbe}
   455→                  disabled={anyBusy(busy)}
   456→                  variant="outline"
   457→                  className="h-7 border-zinc-700 px-2.5 text-[11px] text-zinc-300 hover:bg-zinc-800"
   458→                >
   459→                  {busy === "probe" ? <Loader2 className="mr-1.5 h-3 w-3 animate-spin" /> : <RadioTower className="mr-1.5 h-3 w-3" />}
   460→                  Sondar rotas
   461→                </Button>
   462→              </div>
   463→              <p className={`font-mono text-[10px] leading-relaxed ${routeLabel ? "text-emerald-400" : "text-zinc-500"}`}>
   464→                {routeLabel ?? "não validada — a sonda envia a query de bateria por cada rota candidata; a resposta real do teclado escolhe a rota usada por RGB, desempenho e bateria"}
   465→              </p>
   466→            </div>
   467→          )}
   468→
   469→          <p className="text-xs leading-relaxed text-zinc-500">
   470→            No seletor, marque <strong className="text-zinc-300">todas</strong> as entradas “Aula F75 Max” e “Aula F75 Max 2.4G”.
   471→          </p>
   472→
   473→          <div className="flex flex-wrap gap-2">
   474→            <Button onClick={props.onConnect} disabled={anyBusy(busy) || status.sim} className="h-9 bg-emerald-500 text-zinc-950 hover:bg-emerald-400">
   475→              {busy === "connect" ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Plug className="mr-2 h-4 w-4" />}
   476→              Conectar
   477→            </Button>
   478→            <Button onClick={props.onReconnect} disabled={anyBusy(busy) || status.sim} variant="outline" className="h-9 border-zinc-700 bg-transparent text-zinc-300 hover:bg-zinc-800">
   479→              {busy === "reconnect" ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <RefreshCw className="mr-2 h-4 w-4" />}
   480→              Reconectar
   481→            </Button>
   482→            {anyConnected && (
   483→              <Button onClick={props.onDisconnect} disabled={anyBusy(busy)} variant="outline" className="h-9 border-zinc-700 bg-transparent text-zinc-300 hover:bg-zinc-800">
