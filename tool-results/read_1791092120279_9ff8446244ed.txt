   405→   405→function DevicePanel(props: {
   406→   406→  supported: boolean | null;
   407→   407→  status: DriverStatus;
   408→   408→  anyConnected: boolean;
   409→   409→  wiredReady: boolean;
   410→   410→  busy: string | null;
   411→   411→  battery: number | null;
   412→   412→  driverReady: boolean;
   413→   413→  resetArmed: boolean;
   414→   414→  routeLabel: string | null;
   415→   415→  onConnect: () => void;
   416→   416→  onReconnect: () => void;
   417→   417→  onDisconnect: () => void;
   418→   418→  onToggleSim: () => void;
   419→   419→  onProbe: () => void;
   420→   420→  onFactoryReset: () => void;
   421→   421→}) {
   422→   422→  const { supported, status, anyConnected, wiredReady, busy, battery, driverReady, resetArmed, routeLabel } = props;
