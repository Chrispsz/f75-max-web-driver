    73→
    74→export default function F75App() {
    75→  const driverRef = useRef<F75Driver | null>(null);
    76→  const resetTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    77→
    78→  const [section, setSection] = useState<SectionId>("device");
    79→  const [supported, setSupported] = useState<boolean | null>(null);
    80→  const [inIframe, setInIframe] = useState(false);
    81→  const [driverReady, setDriverReady] = useState(false);
    82→  const [status, setStatus] = useState<DriverStatus>({ wiredCommand: false, wiredDisplay: false, dongle: false });
    83→  const [busy, setBusy] = useState<string | null>(null);
    84→  const [battery, setBattery] = useState<number | null>(null);
    85→  const [routeLabel, setRouteLabel] = useState<string | null>(null);
    86→
    87→  const [rgb, setRgbState] = useState<RgbSettings>({ mode: 1, brightness: 3, speed: 2, direction: 0, colorful: false, color: hexToInt(CYAN) });
    88→  const [perf, setPerfState] = useState<PerfState>({ level: 2, sleep: 2, game: false });
    89→  const [resetArmed, setResetArmed] = useState(false);
    90→
    91→  const setRgb = useCallback((patch: Partial<RgbSettings>) => setRgbState((prev) => ({ ...prev, ...patch })), []);
    92→  const setPerf = useCallback((patch: Partial<PerfState>) => setPerfState((prev) => ({ ...prev, ...patch })), []);
    93→
    94→  /* ------------------------------- bootstrap ------------------------------- */
    95→
    96→  useEffect(() => {
    97→    try {
    98→      const raw = localStorage.getItem(UI_KEY);
    99→      if (raw) {
   100→        const saved = JSON.parse(raw) as { section?: SectionId; rgb?: Partial<RgbSettings>; perf?: Partial<PerfState> };
   101→        if (saved.section && SECTIONS.some((s) => s.id === saved.section)) setSection(saved.section);
   102→        if (saved.rgb) setRgbState((prev) => ({ ...prev, ...saved.rgb }));
   103→        if (saved.perf) setPerfState((prev) => ({ ...prev, ...saved.perf }));
   104→      }
   105→    } catch {
   106→      /* primeira execução */
   107→    }
   108→
   109→    const driver = new F75Driver();
   110→    driverRef.current = driver;
   111→    setDriverReady(true);
   112→    driver.onStatus = (s) => setStatus(s);
   113→    driver.onBattery = (p) => setBattery(p);
   114→    setSupported(F75Driver.supported);
   115→    try {
   116→      setInIframe(window.self !== window.top);
   117→    } catch {
   118→      setInIframe(true);
   119→    }
   120→    f75log.info("F75 Max Web Driver v7 pronto — upload de display em 1 transferência por bloco (wire nativa exata), RGB/desempenho via 2.4G ou cabo, rota validada por sonda. 100% local. Console espelhado aqui e no F12 (filtro [F75]).");
   121→    driver
   122→      .reconnectSaved(true)
   123→      .then((s) => {
   124→        if (s.wiredCommand || s.wiredDisplay || s.dongle) f75log.ok("Dispositivos já autorizados rebinados automaticamente.");
   125→      })
   126→      .catch(() => undefined);
   127→    return () => {
   128→      driver.onStatus = null;
   129→      driver.onBattery = null;
   130→    };
   131→  }, []);
   132→
   133→  useEffect(() => {
   134→    try {
   135→      localStorage.setItem(UI_KEY, JSON.stringify({ section, rgb, perf }));
   136→    } catch {
   137→      /* storage cheio/bloqueado — segue sem persistir */
   138→    }
   139→  }, [section, rgb, perf]);
   140→
   141→  const run = useCallback(
   142→    async (label: string, action: () => Promise<void>) => {
   143→      if (busy) return;
   144→      setBusy(label);
   145→      try {
   146→        await action();
   147→      } catch (err) {
   148→        const message = err instanceof F75Error ? err.message : err instanceof Error ? err.message : String(err);
   149→        f75log.err(`✗ ${label}: ${message}`);
   150→      } finally {
   151→        setBusy(null);
   152→      }
   153→    },
   154→    [busy]
   155→  );
   156→
   157→  /* --------------------------------- ações --------------------------------- */
   158→
   159→  const connect = () =>
   160→    void run("connect", async () => {
   161→      const s = await driverRef.current!.connectPicker();
   162→      if (s.dongle) await driverRef.current!.queryBattery(true).catch(() => undefined);
   163→      setRouteLabel(driverRef.current!.dongleRouteLabel);
   164→    });
   165→
   166→  const reconnect = () =>
   167→    void run("reconnect", async () => {
   168→      const s = await driverRef.current!.reconnectSaved(false);
   169→      if (s.dongle) await driverRef.current!.queryBattery(true).catch(() => undefined);
   170→      setRouteLabel(driverRef.current!.dongleRouteLabel);
   171→    });
   172→
   173→  const probeDongle = () =>
   174→    void run("probe", async () => {
   175→      await driverRef.current!.queryBattery(false);
   176→      setRouteLabel(driverRef.current!.dongleRouteLabel);
   177→    });
   178→
   179→  const disconnect = () =>
   180→    void run("disconnect", async () => {
   181→      setBattery(null);
   182→      await driverRef.current!.disconnect();
   183→    });
   184→
   185→  const applyRgb = () => void run("rgb", () => driverRef.current!.applyRGB(rgb));
   186→
   187→  const applyPerf = () =>
   188→    void run("perf", () => {
   189→      const game = perf.game;
   190→      // Semântica do firmware nativo (setGameMode): game mode liga as quatro
   191→      // flags JUNTAS — Alt+Tab/Alt+F4 o firmware ignora, Win lock é o que
   192→      // produz efeito visível (tecla Win morre + LED branco fixo).
   193→      return driverRef.current!.applyPerformance({
   194→        level: perf.level,
   195→        sleep: perf.sleep,
   196→        game,
   197→        lockAltTab: game,
   198→        lockAltF4: game,
   199→        lockWin: game,
   200→      });
   201→    });
   202→
   203→  const factoryReset = () => {
   204→    if (!resetArmed) {
   205→      setResetArmed(true);
   206→      if (resetTimer.current) clearTimeout(resetTimer.current);
   207→      resetTimer.current = setTimeout(() => setResetArmed(false), 6000);
   208→      return;
   209→    }
   210→    if (resetTimer.current) clearTimeout(resetTimer.current);
   211→    setResetArmed(false);
   212→    void run("reset", async () => {
   213→      await driverRef.current!.factoryReset((stage) => f75log.info(`· ${stage}`));
   214→    });
   215→  };
   216→
   217→  const syncClock = () => void run("clock", () => driverRef.current!.syncClock());
   218→
   219→  /* --------------------------------- estado -------------------------------- */
   220→
   221→  const wiredReady = status.wiredCommand && status.wiredDisplay;
   222→  const anyConnected = status.wiredCommand || status.wiredDisplay || status.dongle;
   223→
   224→  const connectionLabel = status.dongle && wiredReady
   225→    ? "2.4G + Cabo"
   226→    : status.dongle
   227→      ? "Receiver 2.4G"
   228→      : wiredReady
   229→        ? "Cabo USB-C"
   230→        : anyConnected
   231→          ? "Parcial"
   232→          : "Desconectado";
   233→
   234→  return (
   235→    <div className="flex h-dvh flex-col overflow-hidden bg-zinc-950 text-zinc-100 selection:bg-emerald-500/30">
   236→      {/* ------------------------------- topbar ------------------------------- */}
   237→      <header className="flex h-14 shrink-0 items-center gap-3 border-b border-zinc-800/80 px-4 sm:px-5">
   238→        <span className="flex h-8 w-8 items-center justify-center rounded-lg border border-emerald-500/40 bg-emerald-500/10">
   239→          <Keyboard className="h-4 w-4 text-emerald-400" />
   240→        </span>
   241→        <div className="min-w-0">
   242→          <h1 className="truncate text-sm font-bold leading-tight">
   243→            F75 Max <span className="text-emerald-400">Web Driver</span>
   244→          </h1>
   245→          <p className="truncate text-[10px] leading-tight text-zinc-500">protocolo nativo portado · 100% local no navegador</p>
   246→        </div>
   247→
   248→        <div className="ml-auto flex items-center gap-2">
   249→          {status.dongle && (
   250→            <Chip ok={true} title="Bateria do teclado via receiver">
   251→              <BatteryGauge percent={battery} />
   252→            </Chip>
   253→          )}
   254→          <Chip ok={anyConnected} title="Estado da conexão">
   255→            <span className={`h-1.5 w-1.5 rounded-full ${anyConnected ? "bg-emerald-400" : "bg-zinc-600"}`} />
   256→            {connectionLabel}
   257→          </Chip>
   258→        </div>
   259→      </header>
   260→
   261→      <div className="flex min-h-0 flex-1">
   262→        {/* ------------------------------ sidebar ------------------------------ */}
   263→        <aside className="hidden w-56 shrink-0 flex-col justify-between border-r border-zinc-800/80 p-3 md:flex">
   264→          <nav aria-label="Seções do driver" className="space-y-1">
   265→            {SECTIONS.map((s) => (
   266→              <button
   267→                key={s.id}
   268→                type="button"
   269→                onClick={() => setSection(s.id)}
   270→                aria-current={section === s.id ? "page" : undefined}
   271→                className={`flex w-full items-center gap-3 rounded-lg px-3 py-2 text-left text-sm transition-colors ${
   272→                  section === s.id
   273→                    ? "bg-emerald-500/10 font-semibold text-emerald-300"
   274→                    : "text-zinc-400 hover:bg-zinc-900 hover:text-zinc-200"
   275→                }`}
   276→              >
   277→                <s.icon className="h-4 w-4 shrink-0" />
   278→                {s.label}
   279→              </button>
   280→            ))}
   281→          </nav>
   282→          <div className="space-y-2 px-1">
   283→            <div className="flex flex-wrap gap-1.5">
   284→              <Chip ok={supported === true} title={supported ? "WebHID disponível" : "Use Chrome/Chromium/Edge/Brave"}>
   285→                WebHID
   286→              </Chip>
   287→              <Chip title="Nenhuma telemetria — nada sai da máquina">100% local</Chip>
   288→            </div>
   289→            <p className="font-mono text-[10px] leading-relaxed text-zinc-600">v7 · upload de display 1 transferência/bloco · RGB/perf 2.4G ou cabo</p>
   290→          </div>
   291→        </aside>
   292→
   293→        {/* --------------------------- conteúdo ------------------------------ */}
   294→        <div className="flex min-w-0 flex-1 flex-col">
   295→          <nav aria-label="Seções do driver" className="scrollbar-thin flex shrink-0 gap-1 overflow-x-auto border-b border-zinc-800/80 px-3 py-2 md:hidden">
   296→            {SECTIONS.map((s) => (
   297→              <button
   298→                key={s.id}
   299→                type="button"
   300→                onClick={() => setSection(s.id)}
   301→                aria-current={section === s.id ? "page" : undefined}
   302→                className={`flex shrink-0 items-center gap-1.5 rounded-full border px-3 py-1.5 text-xs transition-colors ${
   303→                  section === s.id
   304→                    ? "border-emerald-500/50 bg-emerald-500/10 font-semibold text-emerald-300"
   305→                    : "border-zinc-800 bg-zinc-900/60 text-zinc-400"
   306→                }`}
   307→              >
   308→                <s.icon className="h-3.5 w-3.5" />
   309→                {s.label}
   310→              </button>
   311→            ))}
   312→          </nav>
   313→
   314→          <main className="min-h-0 flex-1 overflow-y-auto p-4 sm:p-6">
   315→            <div className="mx-auto max-w-3xl space-y-5">
   316→              {supported === false && (
   317→                <div className="flex items-start gap-2 rounded-lg border border-amber-500/30 bg-amber-500/5 px-3 py-2.5 text-xs leading-relaxed text-amber-100/90">
   318→                  <TriangleAlert className="mt-0.5 h-4 w-4 shrink-0 text-amber-400" />
   319→                  <span>
   320→                    <strong className="text-amber-300">WebHID indisponível.</strong> Use Chrome, Chromium, Edge, Brave ou Opera (não existe no Firefox).
   321→                  </span>
   322→                </div>
   323→              )}
   324→              {supported && inIframe && (
   325→                <div className="flex items-start gap-2 rounded-lg border border-emerald-500/25 bg-emerald-500/5 px-3 py-2.5 text-xs leading-relaxed text-emerald-100/80">
   326→                  <Cable className="mt-0.5 h-4 w-4 shrink-0 text-emerald-400" />
   327→                  <span>
   328→                    Rodando em iframe: se o seletor de dispositivos não abrir, use <strong>Open in New Tab</strong> acima do preview.
   329→                  </span>
   330→                </div>
   331→              )}
   332→
   333→              {section === "device" && (
   334→                <DevicePanel
   335→                  supported={supported}
   336→                  status={status}
   337→                  anyConnected={anyConnected}
   338→                  wiredReady={wiredReady}
   339→                  busy={busy}
   340→                  battery={battery}
   341→                  driverReady={driverReady}
   342→                  resetArmed={resetArmed}
   343→                  routeLabel={routeLabel}
   344→                  onConnect={connect}
   345→                  onReconnect={reconnect}
   346→                  onDisconnect={disconnect}
   347→                  onProbe={probeDongle}
   348→                  onFactoryReset={factoryReset}
   349→                />
   350→              )}
   351→
   352→              {section === "lighting" && (
   353→                <LightingPanel rgb={rgb} setRgb={setRgb} canSend={anyConnected} busy={busy} onApply={applyRgb} dongle={status.dongle} />
   354→              )}
   355→
   356→              {section === "performance" && <PerformancePanel perf={perf} setPerf={setPerf} canSend={anyConnected} busy={busy} onApply={applyPerf} dongle={status.dongle} />}
   357→
   358→              {section === "display" && <DisplayPanel driver={driverReady ? driverRef.current : null} run={run} busy={busy} ready={wiredReady} />}
   359→
   360→              {section === "keys" && <KeysPanel active={section === "keys"} />}
   361→
   362→              {section === "system" && <SystemPanel driver={driverReady ? driverRef.current : null} />}
   363→            </div>
   364→          </main>
   365→        </div>
   366→      </div>
   367→
   368→      {/* ----------------------------- status bar ----------------------------- */}
   369→      <footer className="shrink-0 border-t border-zinc-800/80 bg-zinc-950">
   370→        <div className="scrollbar-thin flex items-center gap-2 overflow-x-auto px-4 py-2 text-[11px] text-zinc-500 sm:px-5">
   371→          <Chip ok={status.wiredCommand} title="Cabo USB-C · canal de comando 0xFF13">cabo·cmd</Chip>
   372→          <Chip ok={status.wiredDisplay} title="Cabo USB-C · canal de display 0xFF68">cabo·disp</Chip>
   373→          <Chip ok={status.dongle} title="Receiver 2.4G · canais 0xFF59/0xFF60">2.4G</Chip>
   374→          <span className="ml-auto hidden shrink-0 items-center gap-1 font-mono text-[10px] text-zinc-600 sm:flex">
   375→            <Cable className="h-3 w-3" /> 0xFF13 · 0xFF68 · 0xFF60
   376→          </span>
   377→        </div>
   378→      </footer>
   379→    </div>
   380→  );
   381→}

   384→
   385→function DevicePanel(props: {
   386→  supported: boolean | null;
   387→  status: DriverStatus;
   388→  anyConnected: boolean;
   389→  wiredReady: boolean;
   390→  busy: string | null;
   391→  battery: number | null;
   392→  driverReady: boolean;
   393→  resetArmed: boolean;
   394→  routeLabel: string | null;
   395→  onConnect: () => void;
   396→  onReconnect: () => void;
   397→  onDisconnect: () => void;
   398→  onProbe: () => void;
   399→  onFactoryReset: () => void;
   400→}) {
   401→  const { supported, status, anyConnected, wiredReady, busy, battery, driverReady, resetArmed, routeLabel } = props;
   402→
   403→  return (
   404→    <section className="space-y-5" aria-label="Dispositivo">
   405→      <SectionHeader icon={Usb} title="Dispositivo" desc="Conexão e estado do hardware" />
   406→
   407→      <Card className="border-zinc-800 bg-zinc-900/50">
   408→        <CardContent className="space-y-4 p-4 sm:p-5">
   409→          <div className="flex flex-wrap items-center gap-2">
   410→            <Chip ok={status.wiredCommand}>cabo · comando 0xFF13</Chip>
   411→            <Chip ok={status.wiredDisplay}>cabo · display 0xFF68</Chip>
   412→            <Chip ok={status.dongle}>receiver 2.4G</Chip>
   413→            {status.dongle && (
   414→              <Chip ok={battery !== null}>
   415→                <BatteryGauge percent={battery} />
   416→              </Chip>
   417→            )}
   418→          </div>
   419→
   420→          {status.dongle && (
   421→            <div className="space-y-2 rounded-lg border border-zinc-800 bg-zinc-950/60 p-3">
   422→              <div className="flex items-center justify-between gap-2">
   423→                <p className="text-xs font-semibold text-zinc-300">Rota de comando do receiver</p>
   424→                <Button
   425→                  onClick={props.onProbe}
   426→                  disabled={anyBusy(busy)}
   427→                  variant="outline"
   428→                  className="h-7 border-zinc-700 px-2.5 text-[11px] text-zinc-300 hover:bg-zinc-800"
   429→                >
   430→                  {busy === "probe" ? <Loader2 className="mr-1.5 h-3 w-3 animate-spin" /> : <RadioTower className="mr-1.5 h-3 w-3" />}
   431→                  Sondar rotas
   432→                </Button>
   433→              </div>
   434→              <p className={`font-mono text-[10px] leading-relaxed ${routeLabel ? "text-emerald-400" : "text-zinc-500"}`}>
   435→                {routeLabel ?? "não validada — a sonda envia a query de bateria por cada rota candidata; a resposta real do teclado escolhe a rota usada por RGB, desempenho e bateria"}
   436→              </p>
   437→            </div>
   438→          )}
   439→
   440→          <p className="text-xs leading-relaxed text-zinc-500">
   441→            No seletor, marque <strong className="text-zinc-300">todas</strong> as entradas “Aula F75 Max” e “Aula F75 Max 2.4G”.
   442→          </p>
   443→
   444→          <div className="flex flex-wrap gap-2">
   445→            <Button onClick={props.onConnect} disabled={anyBusy(busy)} className="h-9 bg-emerald-500 text-zinc-950 hover:bg-emerald-400">
   446→              {busy === "connect" ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Plug className="mr-2 h-4 w-4" />}
   447→              Conectar
   448→            </Button>
   449→            <Button onClick={props.onReconnect} disabled={anyBusy(busy)} variant="outline" className="h-9 border-zinc-700 bg-transparent text-zinc-300 hover:bg-zinc-800">
   450→              {busy === "reconnect" ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <RefreshCw className="mr-2 h-4 w-4" />}
   451→              Reconectar
   452→            </Button>
   453→            {anyConnected && (
   454→              <Button onClick={props.onDisconnect} disabled={anyBusy(busy)} variant="outline" className="h-9 border-zinc-700 bg-transparent text-zinc-300 hover:bg-zinc-800">
   455→                <PlugZap className="mr-2 h-4 w-4" /> Desconectar
   456→              </Button>
   457→            )}
   458→          </div>
   459→        </CardContent>
   460→      </Card>
   461→
   462→      <Card className="border-rose-500/20 bg-rose-500/[0.03]">
   463→        <CardContent className="space-y-3 p-4 sm:p-5">
   464→          <div className="flex items-center gap-2">
   465→            <TriangleAlert className="h-4 w-4 text-rose-400" />
   466→            <p className="text-sm font-semibold">Restauração de fábrica</p>
   467→          </div>
   468→          <p className="text-xs leading-relaxed text-zinc-500">
   469→            Apaga display, keymap, macros e lighting. Requer cabo USB-C. Dois cliques pra confirmar.
   470→          </p>
   471→          <Button
   472→            onClick={props.onFactoryReset}
   473→            disabled={anyBusy(busy) || !wiredReady}
