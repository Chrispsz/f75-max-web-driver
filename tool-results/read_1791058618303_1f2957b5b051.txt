     1→     1→/**
     2→     2→ * F75 Driver — transporte WebHID do Aula F75 Max (v2 · descriptor-aware).
     3→     3→ *
     4→     4→ * A v1 falhava com NotAllowedError: "Failed to write the report" porque a
     5→     5→ * escrita chegava ao kernel e era rejeitada — a interface errada do receiver
     6→     6→ * era vinculada (a primeira que aparecia, que pode ser a de teclado) e os
     7→     7→ * report IDs não respeitavam o descritor real. Esta versão reproduz a
     8→     8→ * semântica EXATA do trio hidapi/hidraw + Chromium no Linux:
     9→     9→ *
    10→    10→ *  - Chromium SEMPRE prefixa o byte de report ID na wire:
    11→    11→ *      sendReport(id, payload)        → hidraw write([id] + payload)
    12→    12→ *      sendFeatureReport(id, payload) → HIDIOCSFEATURE([id] + payload)
    13→    13→ *  - O Chrome exige has_report_id(interface) === (id !== 0): interface COM
    14→    14→ *    report IDs numerados PRECISA de id ≠ 0; SEM IDs, PRECISA de id 0.
    15→    15→ *  - Kernel (hidraw_send_report): count < 2 → EINVAL; count > 4096
    16→    16→ *    (HID_MAX_BUFFER_SIZE) → EINVAL; no caminho interrupt OUT o conteúdo vai
    17→    17→ *    cru (sem validação de ID); no fallback SET_REPORT o ID é validado no
    18→    18→ *    descritor e o byte de ID é stripado.
    19→    19→ *  - Driver nativo (LinuxHIDBackend.swift):
    20→    20→ *      · receiver 2.4G → hid_write na interface 0xFF60 (raw 32B primeiro,
    21→    21→ *        fallback [0x00]+pacote), candidatos ordenados: 0xFF60 → maior output;
    22→    22→ *      · cabo → feature reports 64B no 0xFF13 (raw primeiro, fallback [0x00]+);
    23→    23→ *      · display → writes de 4096B no 0xFF68.
    24→    24→ *
    25→    25→ * A matriz de estratégias abaixo tenta as equivalentes na ordem que o
    26→    26→ * DESCRITOR real sugere, loga cada tentativa (F12 / painel) e cacheia a
    27→    27→ * vencedora por família de comando. Nada é "adivinhado": cada tentativa
    28→    28→ * fracassada aparece no log com o motivo.
    29→    29→ */

    99→    99→interface ReportSize {
   100→   100→  id: number;
   101→   101→  bytes: number;
   102→   102→}
   103→   103→
   104→   104→interface DeviceSummary {
   105→   105→  usagePages: number[];
   106→   106→  hasNumberedIds: boolean;
   107→   107→  maxOutput: number;
   108→   108→  maxFeature: number;
   109→   109→  outputIds: ReportSize[];
   110→   110→  featureIds: ReportSize[];
   111→   111→}
   112→   112→
   113→   113→function reportBytes(infos: HIDReportInfo[]): ReportSize[] {
   114→   114→  return infos.map((info) => ({
   115→   115→    id: info.reportId,
   116→   116→    bytes: Math.ceil(info.items.reduce((acc, item) => acc + (item.reportCount ?? 1) * (item.reportSize ?? 0), 0) / 8),
   117→   117→  }));
   118→   118→}
   119→   119→
   120→   120→function walkCollections(
   121→   121→  c: HIDCollectionInfo,
   122→   122→  acc: { all: HIDReportInfo[]; out: HIDReportInfo[]; feat: HIDReportInfo[]; pages: Set<number> }
   123→   123→): void {
   124→   124→  acc.pages.add(c.usagePage);
   125→   125→  acc.all.push(...c.inputReports, ...c.outputReports, ...c.featureReports);
   126→   126→  acc.out.push(...c.outputReports);
   127→   127→  acc.feat.push(...c.featureReports);
   128→   128→  for (const child of c.children ?? []) walkCollections(child, acc);
   129→   129→}
   130→   130→
   131→   131→function summarizeDevice(device: HIDDevice): DeviceSummary {
   132→   132→  const acc = { all: [] as HIDReportInfo[], out: [] as HIDReportInfo[], feat: [] as HIDReportInfo[], pages: new Set<number>() };
   133→   133→  for (const c of device.collections) walkCollections(c, acc);
   134→   134→  const outputIds = reportBytes(acc.out);
   135→   135→  const featureIds = reportBytes(acc.feat);
   136→   136→  return {
   137→   137→    usagePages: [...acc.pages],
   138→   138→    hasNumberedIds: acc.all.some((r) => r.reportId !== 0),
   139→   139→    maxOutput: outputIds.reduce((m, r) => Math.max(m, r.bytes), 0),
   140→   140→    maxFeature: featureIds.reduce((m, r) => Math.max(m, r.bytes), 0),
   141→   141→    outputIds,
   142→   142→    featureIds,
   143→   143→  };
   144→   144→}
   145→   145→
   146→   146→function fmtSizes(list: ReportSize[]): string {
   147→   147→  return list.length ? list.map((r) => `0x${hex2(r.id)}:${r.bytes}B`).join(",") : "—";
   148→   148→}
   149→   149→
   150→   150→const padTo = (src: Uint8Array, size: number): Uint8Array => {
   151→   151→  if (src.length >= size) return src;
   152→   152→  const out = new Uint8Array(size);
   153→   153→  out.set(src);
   154→   154→  return out;
   155→   155→};

   270→   270→  /** Abre o seletor do navegador (precisa de gesto do usuário). */
   271→   271→  async connectPicker(): Promise<DriverStatus> {
   272→   272→    if (this.simMode) return this.simConnect();
   273→   273→    if (!navigator.hid) throw new F75Error("WebHID indisponível — use Chrome/Chromium/Edge/Brave (não existe no Firefox).");
   274→   274→    this.attachHidListeners();
   275→   275→
   276→   276→    f75log.info("Abrindo seletor de dispositivos WebHID (filtros: 0x0C45:0x800A cabo, 0x05AC:0x024F receiver)…");
   277→   277→    const devices = await navigator.hid.requestDevice({
   278→   278→      filters: [
   279→   279→        { vendorId: AULA.wiredVendorId, productId: AULA.wiredProductId },
   280→   280→        { vendorId: AULA.dongleVendorId, productId: AULA.dongleProductId },
   281→   281→      ],
   282→   282→    });
   283→   283→    f75log.info(`Seletor retornou ${devices.length} interface(s) — vinculando as do F75 Max.`);
   284→   284→    await this.bindDevices(devices);
   285→   285→    if (!this.wiredCommand && !this.wiredDisplay && this.dongleInterfaces.length === 0) {
   286→   286→      throw new F75Error(
   287→   287→        "Nenhum endpoint vinculado. Selecione TODAS as entradas 'Aula F75 Max' / 'Aula F75 Max 2.4G' no seletor e tente de novo."
   288→   288→      );
   289→   289→    }
   290→   290→    this.notify();
   291→   291→    return this.status;
   292→   292→  }
   293→   293→
   294→   294→  /** Rebinding silencioso dos dispositivos já autorizados (sem seletor). */
   295→   295→  async reconnectSaved(silent = false): Promise<DriverStatus> {
   296→   296→    if (this.simMode) return this.simConnect();
   297→   297→    if (!navigator.hid) throw new F75Error("WebHID indisponível neste navegador.");
   298→   298→    this.attachHidListeners();
   299→   299→    const granted = await navigator.hid.getDevices();
   300→   300→    const ours = granted.filter(
   301→   301→      (d) =>
   302→   302→        (d.vendorId === AULA.wiredVendorId && d.productId === AULA.wiredProductId) ||
   303→   303→        (d.vendorId === AULA.dongleVendorId && d.productId === AULA.dongleProductId)
   304→   304→    );
   305→   305→    if (!silent) f75log.info(`Reconectar: ${ours.length} interface(s) já autorizadas neste navegador.`);
   306→   306→    if (ours.length === 0 && !silent) {
   307→   307→      throw new F75Error("Nenhum dispositivo autorizado ainda — use 'Conectar' para abrir o seletor uma vez.");
   308→   308→    }
   309→   309→    await this.bindDevices(ours);
   310→   310→    this.notify();
   311→   311→    return this.status;
   312→   312→  }

   314→   314→  private async openDevice(device: HIDDevice): Promise<boolean> {
   315→   315→    if (this.bound.has(device)) return false;
   316→   316→    try {
   317→   317→      if (!device.opened) await device.open();
   318→   318→    } catch (err) {
   319→   319→      f75log.err(`Falha ao abrir ${device.productName || "HID"}: ${String(err)} — confira a regra udev (sudo cp packaging/linux/60-aula-f75-max.rules /etc/udev/rules.d/ && sudo udevadm control --reload && replug).`);
   320→   320→      return false;
   321→   321→    }
   322→   322→    this.bound.add(device);
   323→   323→    return true;
   324→   324→  }
   325→   325→
   326→   326→  private makeEndpoint(device: HIDDevice, role: EndpointRole): Endpoint {
   327→   327→    const summary = summarizeDevice(device);
   328→   328→    const ep: Endpoint = { kind: "real", key: `ep${this.keySeq++}`, device, summary, role };
   329→   329→    const pages = summary.usagePages.map((p) => `0x${p.toString(16)}`).join("/");
   330→   330→    f75log.debug(
   331→   331→      `Interface ${role} aberta: ${device.productName || "HID"} · pages ${pages} · reportIds ${summary.hasNumberedIds ? "NUMERADOS" : "ausentes"} · out ${fmtSizes(summary.outputIds)} · feat ${fmtSizes(summary.featureIds)}`
   332→   332→    );
   333→   333→    return ep;
   334→   334→  }
   335→   335→
   336→   336→  private async bindDevices(devices: HIDDevice[]): Promise<void> {
   337→   337→    const dongleCandidates: HIDDevice[] = [];
   338→   338→
   339→   339→    for (const device of devices) {
   340→   340→      const isWired = device.vendorId === AULA.wiredVendorId && device.productId === AULA.wiredProductId;
   341→   341→      const isDongle = device.vendorId === AULA.dongleVendorId && device.productId === AULA.dongleProductId;
   342→   342→      if ((!isWired && !isDongle) || this.bound.has(device)) continue;
   343→   343→
   344→   344→      if (isDongle) {
   345→   345→        dongleCandidates.push(device);
   346→   346→        continue;
   347→   347→      }
   348→   348→
   349→   349→      const role: EndpointRole | null = device.collections.some((c) => c.usagePage === AULA.wiredCommandPage)
   350→   350→        ? "wiredCommand"
   351→   351→        : device.collections.some((c) => c.usagePage === AULA.wiredRawPage)
   352→   352→          ? "wiredDisplay"
   353→   353→          : null;
   354→   354→      if (!role) {
   355→   355→        f75log.debug(`Interface do cabo ignorada (sem usage page 0xFF13/0xFF68): ${device.productName || "HID"}`);
   356→   356→        continue;
   357→   357→      }
   358→   358→      if (this[role]) continue; // já vinculada
   359→   359→      if (!(await this.openDevice(device))) continue;
   360→   360→      const ep = this.makeEndpoint(device, role);
   361→   361→      this[role] = ep;
   362→   362→      ep.device.addEventListener("inputreport", (ev) =>
   363→   363→        this.handleInput(role, ev.reportId, new Uint8Array(ev.data.buffer, ev.data.byteOffset, ev.data.byteLength))
   364→   364→      );
   365→   365→      f75log.ok(`✔ ${role === "wiredCommand" ? "Cabo · comando 0xFF13" : "Cabo · display 0xFF68"} vinculado — ${device.productName || "HID"}`);
   366→   366→    }
   367→   367→
   368→   368→    /* Receiver 2.4G: o nativo (openDongleRaw) varre TODAS as interfaces do
   369→   369→     * 05AC:024F e ordena: usagePage 0xFF60 primeiro, depois maior output.
   370→   370→     * A v1 pegava a primeira interface que aparecia — que pode ser a de
   371→   371→     * teclado do receiver, sem output de 32B → NotAllowedError. */
   372→   372→    for (const device of dongleCandidates) {
   373→   373→      if (!(await this.openDevice(device))) continue;
   374→   374→      const ep = this.makeEndpoint(device, "dongle");
   375→   375→      this.dongleInterfaces.push(ep);
   376→   376→      ep.device.addEventListener("inputreport", (ev) =>
   377→   377→        this.handleInput("dongle", ev.reportId, new Uint8Array(ev.data.buffer, ev.data.byteOffset, ev.data.byteLength))
   378→   378→      );
   379→   379→    }
   380→   380→    if (this.dongleInterfaces.length > 0) {
   381→   381→      this.dongleInterfaces.sort((a, b) => {
   382→   382→        const aRaw = a.summary.usagePages.includes(AULA.dongleRawPage) ? 0 : 1;
   383→   383→        const bRaw = b.summary.usagePages.includes(AULA.dongleRawPage) ? 0 : 1;
   384→   384→        if (aRaw !== bRaw) return aRaw - bRaw;
   385→   385→        return b.summary.maxOutput - a.summary.maxOutput;
   386→   386→      });
   387→   387→      this.dongle = this.dongleInterfaces[0];
   388→   388→      f75log.ok(
   389→   389→        `✔ Receiver 2.4G vinculado em ${this.dongleInterfaces.length} interface(s) — canal preferido: ${this.dongle.summary.usagePages.map((p) => `0x${p.toString(16)}`).join("/")} (output ${this.dongle.summary.maxOutput}B).`
   390→   390→      );
   391→   391→      if (!this.dongle.summary.usagePages.includes(AULA.dongleRawPage)) {
   392→   392→        f75log.warn("Nenhuma interface do receiver expôs a usage page 0xFF60 — usando a de maior output. Se RGB falhar, reconecte com o cabo e refaça o pareamento (Fn+R).");
   393→   393→      }
   394→   394→    }
   395→   395→  }
   396→   396→
   397→   397→  async disconnect(): Promise<void> {
   398→   398→    if (this.simMode) {
   399→   399→      this.simMode = false;
   400→   400→      f75log.info("🧪 Simulação encerrada.");
   401→   401→      this.clearRealEndpoints();
   402→   402→      this.notify();
   403→   403→      return;
   404→   404→    }
   405→   405→    const all = [this.wiredCommand, this.wiredDisplay, ...this.dongleInterfaces];
   406→   406→    for (const ep of all) {
   407→   407→      if (ep?.kind === "real") {
   408→   408→        try {
   409→   409→          await ep.device.close();
   410→   410→        } catch {
   411→   411→          /* ignore */
   412→   412→        }
   413→   413→        this.bound.delete(ep.device);
   414→   414→      }
   415→   415→    }
   416→   416→    this.clearRealEndpoints();
   417→   417→    f75log.info("Desconectado (endpoints fechados). Autorização continua salva — use Reconectar.");
   418→   418→    this.notify();
   419→   419→  }

   445→   445→  private handleInput(role: EndpointRole, reportId: number, data: Uint8Array): void {
   446→   446→    if (role === "dongle") {
   447→   447→      const percent = parseBatteryReport(reportId, data);
   448→   448→      if (percent !== null) {
   449→   449→        f75log.rx(`🔋 Bateria via input report: ${percent}% (id=0x${hex2(reportId)})`);
   450→   450→        const waiters = [...this.batteryWaiters];
   451→   451→        this.batteryWaiters = [];

   453→   453→        this.onBattery?.(percent);

   455→   455→      }
   456→   456→      // teclado/mouse padrão do receiver — ruído de digitação, não loga
   457→   457→      if (reportId === 0x00 || reportId === 0x01 || reportId === 0x02) return;
   458→   458→      f75log.debug(`Input do receiver (id=0x${hex2(reportId)}): ${Array.from(data.subarray(0, 16), (b) => hex2(b)).join(" ")}`);
   459→   459→      return;
   460→   460→    }
   461→   461→    if (role === "wiredDisplay") {
   462→   462→      this.displayAckCounter += 1;
   463→   463→      if (this.displayAckCounter === 1 || this.displayAckCounter % 64 === 0) {
   464→   464→        f75log.rx(`ACK display #${this.displayAckCounter} (id=0x${hex2(reportId)}, ${data.length}B)`);
   465→   465→      }
   466→   466→      return;
   467→   467→    }
   468→   468→    if (role === "wiredCommand") {
   469→   469→      f75log.rx(`Input no canal de comando (id=0x${hex2(reportId)}): ${Array.from(data.subarray(0, 16), (b) => hex2(b)).join(" ")}`);
   470→   470→    }
   471→   471→  }

   473→   473→  private waitForInput(ep: Endpoint | undefined, timeoutMs: number): Promise<Uint8Array | null> {
   474→   474→    if (!ep) return Promise.resolve(null);
   475→   475→    if (this.simMode) {
   476→   476→      return sleep(Math.min(timeoutMs, 25)).then(() => new Uint8Array([0x01]));
   477→   477→    }
   478→   478→    if (ep.kind !== "real") return Promise.resolve(null);
   479→   479→    return new Promise((resolve) => {
   480→   480→      let done = false;
   481→   481→      const finish = (value: Uint8Array | null) => {
   482→   482→        if (done) return;
   483→   483→        done = true;
   484→   484→        clearTimeout(timer);
   485→   485→        ep.device.removeEventListener("inputreport", handler);
   486→   486→        resolve(value);
   487→   487→      };
   488→   488→      const handler = (ev: HIDInputReportEvent) =>
   489→   489→        finish(new Uint8Array(ev.data.buffer, ev.data.byteOffset, ev.data.byteLength));
   490→   490→      const timer = setTimeout(() => finish(null), timeoutMs);
   491→   491→      ep.device.addEventListener("inputreport", handler);
   492→   492→    });
   493→   493→  }

   497→   497→  /**
   498→   498→   * Constrói a matriz de tentativas para um pacote, na ordem sugerida pelo
   499→   499→   * descritor REAL da interface (equivalentes exatas das duas rotas do
   500→   500→   * nativo: raw direto e [0x00]-prefixado).
   501→   501→   */
   502→   502→  private buildAttempts(s: DeviceSummary, packet: Uint8Array, modes: WireMode[]): TxSpec[] {
   503→   503→    const attempts: TxSpec[] = [];
   504→   504→    const numbered = s.hasNumberedIds;
   505→   505→
   506→   506→    for (const mode of modes) {
   507→   507→      const max = mode === "output" ? s.maxOutput : s.maxFeature;
   508→   508→      const declared = mode === "output" ? s.outputIds : s.featureIds;
   509→   509→      if (max <= 0) {
   510→   510→        attempts.push({
   511→   511→          mode,
   512→   512→          reportId: -1,
   513→   513→          payloadLen: 0,
   514→   514→          padded: false,
   515→   515→          note: `PULADO — interface não declara ${mode === "output" ? "output" : "feature"} reports (o Chrome rejeita na hora)`,
   516→   516→        });
   517→   517→        continue;
   518→   518→      }
   519→   519→      const wire = (payloadLen: number) => payloadLen + 1;
   520→   520→      if (numbered) {
   521→   521→        const id = packet[0];
   522→   522→        if (id !== 0) {
   523→   523→          const payloadLen = packet.length - 1;
   524→   524→          if (payloadLen <= max) {
   525→   525→            attempts.push({
   526→   526→              mode,
   527→   527→              reportId: id,
   528→   528→              payloadLen,
   529→   529→              padded: false,
   530→   530→              note: `id=0x${hex2(id)} + ${payloadLen}B payload → ${wire(payloadLen)}B na wire (device numerado — rota "raw" do nativo)`,
   531→   531→            });
   532→   532→          }
   533→   533→          const decl = declared.find((r) => r.id === id);
   534→   534→          if (decl && decl.bytes - 1 > payloadLen && decl.bytes - 1 <= max) {
   535→   535→            attempts.push({
   536→   536→              mode,
   537→   537→              reportId: id,
   538→   538→              payloadLen: decl.bytes - 1,
   539→   539→              padded: true,
   540→   540→              note: `id=0x${hex2(id)} + payload padded a ${decl.bytes - 1}B (report 0x${hex2(id)} declara ${decl.bytes}B)`,
   541→   541→            });
   542→   542→          }
   543→   543→        }
   544→   544→      } else {
   545→   545→        if (packet.length <= max) {
   546→   546→          attempts.push({
   547→   547→            mode,
   548→   548→            reportId: 0,
   549→   549→            payloadLen: packet.length,
   550→   550→            padded: false,
   551→   551→            note: `id=0 + pacote ${packet.length}B → ${wire(packet.length)}B na wire (device sem report IDs — rota prefixed do nativo)`,
   552→   552→          });
   553→   553→        }
   554→   554→        if (max > packet.length && max <= 4095) {
   555→   555→          attempts.push({
   556→   556→            mode,
   557→   557→            reportId: 0,
   558→   558→            payloadLen: max,
   559→   559→            padded: true,
   560→   560→            note: `id=0 + pacote padded a ${max}B (maior ${mode} declarado)`,
   561→   561→          });
   562→   562→        }
   563→   563→      }
   564→   564→    }
   565→   565→
   566→   566→    if (attempts.every((a) => a.reportId === -1)) {
   567→   567→      // Último recurso cruzado — gera evidência no log mesmo sabendo que o
   568→   568→      // Chrome deve rejeitar (divergência has_report_id).
   569→   569→      const mode: WireMode = s.maxOutput > 0 ? "output" : "feature";
   570→   570→      const max = mode === "output" ? s.maxOutput : s.maxFeature;
   571→   571→      if (max > 0) {
   572→   572→        if (numbered && packet[0] !== 0) {
   573→   573→          attempts.push({ mode, reportId: packet[0], payloadLen: packet.length - 1, padded: false, note: "cruzado: id=primeiro byte" });
   574→   574→        } else {
   575→   575→          attempts.push({ mode, reportId: 0, payloadLen: packet.length, padded: false, note: "cruzado: id=0 (esperado falhar — evidência)" });
   576→   576→        }
   577→   577→      }
   578→   578→    }
   579→   579→    return attempts;
   580→   580→  }

   597→   597→  private async emit(
   598→   598→    ep: Endpoint,
   599→   599→    spec: TxSpec,
   600→   600→    packet: Uint8Array,
   601→   601→    label: string,
   602→   602→    logs: TxAttemptLog[],
   603→   603→    dump: boolean
   604→   604→  ): Promise<boolean> {
   605→   605→    const target = `${ep.role}·${ep.summary.usagePages.map((p) => `0x${p.toString(16)}`).join("/")}`;
   606→   606→    if (spec.reportId === -1) {
   607→   607→      logs.push({ target, mode: spec.mode, reportId: "—", wire: "—", result: spec.note });
   608→   608→      f75log.debug(`TX ${spec.mode} · ${label} · ${spec.note}`);
   609→   609→      return false;
   610→   610→    }
   611→   611→    const payload = spec.reportId === 0
   612→   612→      ? spec.padded
   613→   613→        ? padTo(packet, spec.payloadLen)
   614→   614→        : packet
   615→   615→      : spec.padded
   616→   616→        ? padTo(packet.subarray(1), spec.payloadLen)
   617→   617→        : packet.subarray(1);
   618→   618→    const wire = payload.length + 1;
   619→   619→    const idTxt = `0x${hex2(spec.reportId)}`;
   620→   620→    logs.push({
   621→   621→      target,
   622→   622→      mode: spec.mode,
   623→   623→      reportId: idTxt,
   624→   624→      wire: `${payload.length}B+id`,
   625→   625→      result: "…",
   626→   626→    });
   627→   627→    f75log.cmd(`TX ${spec.mode} id=${idTxt} · ${label} · ${payload.length}B payload / ${wire}B na wire · ${spec.note}`);
   628→   628→    if (dump) f75log.dump(payload, `TX ${spec.mode} id=${idTxt} · ${label}`);
   629→   629→    else f75log.debug(`TX ${spec.mode} id=${idTxt} · ${label} · primeiros bytes: ${Array.from(payload.subarray(0, 12), (b) => hex2(b)).join(" ")}`);
   630→   630→    try {
   631→   631→      if (spec.mode === "output") {
   632→   632→        await ep.device.sendReport(spec.reportId, payload);
   633→   633→      } else {
   634→   634→        await ep.device.sendFeatureReport(spec.reportId, payload);
   635→   635→      }
   636→   636→      logs[logs.length - 1].result = "✔ ok";
   637→   637→      f75log.debug(`✔ ${spec.mode} id=${idTxt} aceito pelo SO · ${label}`);
   638→   638→      return true;
   639→   639→    } catch (err) {
   640→   640→      const msg = err instanceof Error ? `${err.name}: ${err.message}` : String(err);
   641→   641→      logs[logs.length - 1].result = `✗ ${msg}`;
   642→   642→      f75log.warn(`${spec.mode} id=${idTxt} falhou · ${label} · ${msg}`);
   643→   643→      return false;
   644→   644→    }
   645→   645→  }

   647→   647→  /**
   648→   648→   * Envia um pacote pela matriz de estratégias. Primeiro tenta a tentativa
   649→   649→   * cacheada (txKey), depois varre candidatos × modos. Primeiro sucesso
   650→   650→   * vence; tudo falhou → F75Error com resumo.
   651→   651→   */
   652→   652→  private async tx(
   653→   653→    packet: Uint8Array,
   654→   654→    label: string,
   655→   655→    role: EndpointRole,
   656→   656→    modes: WireMode[] = ["output", "feature"],
   657→   657→    txKey?: string,
   658→   658→    dump = true
   659→   659→  ): Promise<void> {
   660→   660→    if (this.simMode) {
   661→   661→      f75log.cmd(`TX (sim) · ${label} · ${role === "dongle" ? describeWireless(packet) : describeWired(packet)}`);
   662→   662→      if (dump) f75log.dump(packet, `TX sim · ${label}`);
   663→   663→      return;
   664→   664→    }
   665→   665→
   666→   666→    const candidates = this.candidatesFor(role);
   667→   667→    if (candidates.length === 0) throw new F75Error(this.requireMessage(role));
   668→   668→
   669→   669→    const logs: TxAttemptLog[] = [];
   670→   670→
   671→   671→    if (txKey) {
   672→   672→      const cached = this.txCache.get(txKey);
   673→   673→      if (cached) {
   674→   674→        const ep = candidates.find((c) => c.key === cached.endpointKey);
   675→   675→        if (ep) {
   676→   676→          const ok = await this.emit(ep, cached.spec, packet, label, logs, dump);
   677→   677→          if (ok) {
   678→   678→            this.lastTx = { label, attempts: logs };
   679→   679→            return;
   680→   680→          }
   681→   681→        }
   682→   682→        this.txCache.delete(txKey);
   683→   683→        f75log.warn(`Estratégia cacheada para ${txKey} deixou de funcionar — revarrendo matriz…`);
   684→   684→      }
   685→   685→    }
   686→   686→
   687→   687→    for (const ep of candidates) {
   688→   688→      for (const spec of this.buildAttempts(ep.summary, packet, modes)) {
   689→   689→        const ok = await this.emit(ep, spec, packet, label, logs, dump);
   690→   690→        if (ok) {
   691→   691→          if (txKey) this.txCache.set(txKey, { endpointKey: ep.key, spec });
   692→   692→          this.lastTx = { label, attempts: logs };
   693→   693→          return;
   694→   694→        }
   695→   695→      }
   696→   696→    }
   697→   697→
   698→   698→    this.lastTx = { label, attempts: logs };
   699→   699→    const summary = logs
   700→   700→      .filter((l) => l.result !== "…" && !l.result.startsWith("✔"))
   701→   701→      .map((l) => `${l.target}/${l.mode}/id=${l.reportId}: ${l.result}`)
   702→   702→      .join(" · ");
   703→   703→    throw new F75Error(`TX falhou em ${logs.length} tentativa(s) — ${label}. ${this.requireMessage(role)} Detalhe: ${summary || "nenhuma estratégia aplicável ao descritor"}`);
   704→   704→  }

   706→   706→  /** SET feature + GET feature de ACK — mesma tolerância do commandExchange nativo. */
   707→   707→  private async commandExchange(packet: Uint8Array, label: string): Promise<void> {
   708→   708→    await this.tx(packet, label, "wiredCommand", ["feature", "output"], `wired-${hex2(packet[0])}-${hex2(packet[1])}`);
   709→   709→    await this.readFeatureAck(label);
   710→   710→  }
   711→   711→
   712→   712→  private async readFeatureAck(label: string): Promise<void> {
   713→   713→    if (this.simMode) return;
   714→   714→    const ep = this.wiredCommand;
   715→   715→    if (!ep) return;
   716→   716→    const declared = ep.summary.featureIds.map((r) => r.id);
   717→   717→    const ids = [...new Set([0, ...declared, 0x04])];
   718→   718→    for (const id of ids) {
   719→   719→      try {
   720→   720→        const view = await ep.device.receiveFeatureReport(id);
   721→   721→        f75log.rx(`GET feature id=0x${hex2(id)} · ${label} · ${view.byteLength}B`);
   722→   722→        f75log.dump(view, `GET feature · ${label}`);
   723→   723→        return;
   724→   724→      } catch {
   725→   725→        /* tenta próximo id */
   726→   726→      }
   727→   727→    }
   728→   728→    f75log.debug(`GET feature indisponível (${label}) — seguindo sem ACK (o conteúdo não é validado nem pelo nativo).`);
   729→   729→  }

   744→   744→  /* --------------------------------- display -------------------------------- */
   745→   745→
   746→   746→  /**
   747→   747→   * Bloco de 4 KB do display. O nativo faz hid_write(4096) cru — o kernel
   748→   748→   * aceita (≤ HID_MAX_BUFFER_SIZE) porque interrupt OUT não valida ID. O
   749→   749→   * WebHID SEMPRE injeta o byte de ID, então:
   750→   750→   *  - interface numerada  → sendReport(chunk[0], chunk[1..]) recria a wire
   751→   751→   *    nativa 4096B byte a byte;
   752→   752→   *  - interface sem IDs   → sendReport(0, pedaço) com pedaços ≤ min(maxOut,
   753→   753→   *    4095); a wire ganha um 0x00 por write (o firmware aceita framing
   754→   754→   *    prefixed — o nativo usa esse caminho como fallback).
   755→   755→   */
   756→   756→  private async writeDisplayChunk(ep: Endpoint | undefined, chunk: Uint8Array, index: number, total: number): Promise<void> {
   757→   757→    if (this.simMode) {
   758→   758→      if (index === 1 || index % 64 === 0) f75log.cmd(`TX (sim) · bloco display ${index}/${total} · ${chunk.length}B`);
   759→   759→      return;
   760→   760→    }
   761→   761→    if (!ep) throw new F75Error(this.requireMessage("wiredDisplay"));
   762→   762→    const s = ep.summary;
   763→   763→    if (s.maxOutput <= 0 && s.maxFeature <= 0) {
   764→   764→      throw new F75Error("Interface de display não declara output/feature reports — impossível enviar blocos.");
   765→   765→    }
   766→   766→    if (s.hasNumberedIds && s.maxOutput >= chunk.length - 1) {
   767→   767→      // wire nativa idêntica: [chunk[0]] + chunk[1..4095] = 4096B
   768→   768→      await ep.device.sendReport(chunk[0], chunk.subarray(1));
   769→   769→      if (index === 1) f75log.ok("Estratégia de display: monolítica numerada (wire idêntica ao nativo: [chunk[0]]+4095B).");
   770→   770→      return;
   771→   771→    }
   772→   772→    const piece = Math.min(s.maxOutput > 0 ? s.maxOutput : s.maxFeature, 4095);
   773→   773→    if (piece < 2) throw new F75Error(`Interface de display declara reports muito pequenos (${piece}B).`);
   774→   774→    if (index === 1) {
   775→   775→      f75log.info(
   776→   776→        `Estratégia de display: stream em pedaços de ${piece}B com id=0 (interface sem report IDs; wire = [0x00]+pedaço — kernel limita a 4096B).`
   777→   777→      );
   778→   778→    }
   779→   779→    const useFeature = s.maxOutput <= 0;
   780→   780→    for (let off = 0; off < chunk.length; off += piece) {
   781→   781→      const slice = chunk.subarray(off, Math.min(off + piece, chunk.length));
   782→   782→      if (useFeature) await ep.device.sendFeatureReport(0, slice);
   783→   783→      else await ep.device.sendReport(0, slice);
   784→   784→    }
   785→   785→  }
   786→   786→
   787→   787→  async uploadDisplay(
   788→   788→    stream: EncodedDisplayStream,
   789→   789→    slot: number,
   790→   790→    onProgress: (p: UploadProgress) => void,
   791→   791→    token?: CancelToken
   792→   792→  ): Promise<void> {
   793→   793→    if (!Number.isInteger(slot) || slot < 1 || slot > 255) {
   794→   794→      throw new F75Error("Slot precisa ser um número entre 1 e 255.");
   795→   795→    }
   796→   796→    if (stream.chunkCount > 0xffff) throw new F75Error("Payload grande demais pros metadados de 16 bits.");
   797→   797→    if (!this.simMode) {
   798→   798→      if (!this.wiredCommand) throw new F75Error(this.requireMessage("wiredCommand"));
   799→   799→      if (!this.wiredDisplay) throw new F75Error(this.requireMessage("wiredDisplay"));
   800→   800→    }
   801→   801→
   802→   802→    const raw = this.simMode ? undefined : this.wiredDisplay!;
   803→   803→    const started = performance.now();
   804→   804→    this.displayAckCounter = 0;
   805→   805→
   806→   806→    f75log.info(`📤 Upload slot ${slot}: ${stream.frameCount} frames · ${stream.chunkCount} blocos · ${(stream.data.length / 1024).toFixed(0)} KB · ${stream.avgFps.toFixed(1)} fps médios`);
   807→   807→    await this.commandExchange(wiredPacket(0x04, 0x18), "upload · abrir sessão");
   808→   808→
   809→   809→    const metadata = wiredPacket(0x04, 0x72);
   810→   810→    metadata[2] = slot;
   811→   811→    metadata[8] = stream.chunkCount & 0xff;
   812→   812→    metadata[9] = (stream.chunkCount >> 8) & 0xff;
   813→   813→    await this.commandExchange(metadata, "upload · metadados");
   814→   814→    const firstAck = await this.waitForInput(raw, 150);
   815→   815→    if (!firstAck) f75log.warn("Sem ACK após metadados (150 ms) — o nativo segue mesmo assim.");
   816→   816→
   817→   817→    let lastLog = 0;
   818→   818→    for (let index = 0; index < stream.chunkCount; index++) {
   819→   819→      if (token?.cancelled) throw new F75Error("Upload cancelado pelo usuário.");
   820→   820→      const chunk = stream.data.subarray(index * AULA.chunkLength, (index + 1) * AULA.chunkLength);
   821→   821→      try {
   822→   822→        await this.writeDisplayChunk(raw, chunk, index + 1, stream.chunkCount);
   823→   823→      } catch (err) {
   824→   824→        throw new F75Error(`Bloco ${index + 1}/${stream.chunkCount} falhou: ${err instanceof Error ? err.message : String(err)}`);
   825→   825→      }
   826→   826→      const ack = await this.waitForInput(raw, 350);
   827→   827→      if (!ack && index % 32 === 0) f75log.debug(`Bloco ${index + 1}: sem ACK em 350 ms (seguindo).`);
   828→   828→
   829→   829→      if (index - lastLog >= 9 || index === stream.chunkCount - 1) {
   830→   830→        lastLog = index;
   831→   831→        const elapsed = (performance.now() - started) / 1000;
   832→   832→        const cps = (index + 1) / Math.max(elapsed, 0.001);
   833→   833→        f75log.debug(`Progresso: ${index + 1}/${stream.chunkCount} blocos · ${cps.toFixed(0)} blocos/s`);
   834→   834→        onProgress({
   835→   835→          sent: index + 1,
   836→   836→          total: stream.chunkCount,
   837→   837→          chunksPerSecond: cps,
   838→   838→          etaSeconds: (stream.chunkCount - index - 1) / Math.max(cps, 0.001),
   839→   839→        });
   840→   840→      }
   841→   841→    }
   842→   842→    onProgress({ sent: stream.chunkCount, total: stream.chunkCount, chunksPerSecond: stream.chunkCount / ((performance.now() - started) / 1000), etaSeconds: 0 });
   843→   843→
   844→   844→    await this.commandExchange(wiredPacket(0x04, 0x02), "upload · commit");
   845→   845→    const seconds = (performance.now() - started) / 1000;
   846→   846→    f75log.ok(`Upload concluído no slot ${slot}: ${stream.chunkCount} blocos em ${seconds.toFixed(1)} s (${(stream.chunkCount / Math.max(seconds, 0.001)).toFixed(0)} blocos/s).`);
   847→   847→  }

   922→   922→  async queryBattery(): Promise<number | null> {
   923→   923→    if (this.simMode) {
   924→   924→      f75log.cmd("TX (sim) · battery query");
   925→   925→      await sleep(300);
   926→   926→      f75log.rx("🔋 Bateria via input report: 87% (sim)");

   928→   928→      return 87;
   929→   929→    }
   930→   930→    if (this.dongleInterfaces.length === 0) throw new F75Error(this.requireMessage("dongle"));
   931→   931→    const preferred = this.dongle;
   932→   932→    const maxOut = preferred?.summary.maxOutput ?? 32;
   933→   933→    const lengths = [...new Set([Math.min(Math.max(maxOut, 32), 64), 33, 32])].filter((l) => l <= Math.max(maxOut, 32)).sort((a, b) => b - a);

   937→   937→      new Promise<number | null>((resolve) => {

   940→   940→          const i = this.batteryWaiters.indexOf(resolve);
   941→   941→          if (i >= 0) {
   942→   942→            this.batteryWaiters.splice(i, 1);
   943→   943→            resolve(null);
   944→   944→          }
   945→   945→        }, 250);
   946→   946→      });
   947→   947→
   948→   948→    for (const length of lengths) {
   949→   949→      for (const withId of [false, true]) {
   950→   950→        await this.tx(batteryQueryPacket(withId, length), `battery query ${length}B id=${withId ? "sim" : "não"}`, "dongle", ["output", "feature"], `batt-${length}-${withId ? 1 : 0}`, false);
