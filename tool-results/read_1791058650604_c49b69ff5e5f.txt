   214→const LOOP_SCRIPT = `#!/usr/bin/env python3
   215→# loop_perfeito.py — dissolve o final de volta pro 1º frame = loop sem pulo visível
   216→# uso: python3 loop_perfeito.py entrada.gif saida.gif
   217→# deps: sudo pacman -S python-pillow python-numpy
   218→from PIL import Image, ImageSequence
   219→import numpy as np, sys
   220→
   221→src, dst = sys.argv[1], sys.argv[2]
   222→im = Image.open(src)
   223→dur = im.info.get("duration", 100)  # ms por frame
   224→fr = np.stack([np.asarray(f.convert("RGB"), np.float32) for f in ImageSequence.Iterator(im)])
   225→L, T = len(fr), max(1, int(len(fr) * 0.12))  # dissolve = 12% da duração
   226→out = []
   227→for i in range(L):
   228→    f = fr[i]
   229→    if i >= L - T:
   230→        w = (i - (L - T) + 1) / T
   231→        f = (1 - w) * f + w * fr[0]  # mistura progressiva pro 1º frame
   232→    out.append(Image.fromarray(np.clip(f, 0, 255).astype(np.uint8)))
   233→out[0].save(dst, save_all=True, append_images=out[1:], loop=0, duration=dur)
   234→print(f"ok: {L} frames, dissolve de {T} — último frame == primeiro (seam 0)")`;

   308→function CodeBlock({ code }: { code: string }) {
   309→  const { toast } = useToast();
   310→  const [copied, setCopied] = useState(false);
   311→
   312→  const copy = async () => {
   313→    try {
   314→      await navigator.clipboard.writeText(code);
   315→      setCopied(true);
   316→      toast({ title: "Copiado! 📋", description: "Cole no terminal com Ctrl+Shift+V" });
   317→      setTimeout(() => setCopied(false), 2000);
   318→    } catch {
   319→      toast({ title: "Erro ao copiar", description: "Selecione e copie manualmente" });
   320→    }
   321→  };
   322→
   323→  return (
   324→    <div className="group relative rounded-lg border border-zinc-700/80 bg-zinc-950">
   325→      <Button
   326→        size="sm"
   327→        variant="ghost"
   328→        onClick={copy}
   329→        aria-label="Copiar comando"
   330→        className="absolute right-2 top-2 h-8 w-8 p-0 text-zinc-400 hover:bg-zinc-800 hover:text-emerald-400"
   331→      >
   332→        {copied ? <Check className="h-4 w-4 text-emerald-400" /> : <Copy className="h-4 w-4" />}

   334→      <pre className="overflow-x-auto p-4 pr-12 text-[13px] leading-relaxed text-zinc-200 scrollbar-thin">
   335→        <code className="font-mono">{code}</code>
   336→      </pre>
   337→    </div>
   338→  );
   339→}
