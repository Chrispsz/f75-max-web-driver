    53→/* ------------------------------- copy block ------------------------------- */
    54→
    55→export function CopyBlock({ title, code }: { title?: string; code: string }) {
    56→  const [copied, setCopied] = useState(false);
    57→  const copy = async () => {
    58→    try {
    59→      await navigator.clipboard.writeText(code);
    60→      setCopied(true);
    61→      setTimeout(() => setCopied(false), 1600);
    62→    } catch {
    63→      f75log.err("Não foi possível copiar (clipboard bloqueado) — selecione o texto manualmente.");
    64→    }
    65→  };
    66→  return (
    67→    <div className="overflow-hidden rounded-lg border border-zinc-800 bg-zinc-950">
    68→      <div className="flex items-center justify-between border-b border-zinc-800 bg-zinc-900/80 px-3 py-1.5">
    69→        <span className="font-mono text-[11px] text-zinc-500">{title ?? "bash"}</span>
    70→        <Button variant="ghost" size="sm" onClick={copy} className="h-7 gap-1.5 px-2 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-emerald-400">
    71→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <ClipboardCopy className="h-3.5 w-3.5" />}
    72→          {copied ? "Copiado!" : "Copiar"}
    73→        </Button>
    74→      </div>
    75→      <pre className="scrollbar-thin max-h-72 overflow-auto p-3 font-mono text-xs leading-relaxed text-zinc-300">{code}</pre>
    76→    </div>
    77→  );
    78→}

   336→export function DisplayCard({
   337→  wiredConnected,
   338→  busy,
   339→  prepared,
   340→  onPrepare,
   341→  onGenerate,
   342→  onClear,
   343→  slot,
   344→  setSlot,
   345→  fit,
   346→  setFit,
   347→  progress,
   348→  uploading,
   349→  onCancelUpload,
   350→  onUpload,
   351→  onClock,
   352→  autoClock,
   353→  setAutoClock,
   354→  onFactoryReset,
   355→  resetArmed,
   356→}: {
   357→  wiredConnected: boolean;
   358→  busy: string | null;
   359→  prepared: PreparedUpload | null;
   360→  onPrepare: (file: File | null) => void;
   361→  onGenerate: (kind: "bounce" | "plasma") => void;
   362→  onClear: () => void;
   363→  slot: string;
   364→  setSlot: (v: string) => void;
   365→  fit: FitMode;
   366→  setFit: (f: FitMode) => void;
   367→  progress: { sent: number; total: number; eta: number };
   368→  uploading: boolean;
   369→  onCancelUpload: () => void;
   370→  onUpload: () => void;
   371→  onClock: () => void;
   372→  autoClock: boolean;
   373→  setAutoClock: (v: boolean) => void;
   374→  onFactoryReset: () => void;
   375→  resetArmed: boolean;
   376→}) {
   377→  const previewRef = useRef<HTMLCanvasElement | null>(null);
   378→  const [showDeviceColors, setShowDeviceColors] = useState(false);
   379→
   380→  /* animação da prévia (frames originais ou o RGB565 que a telinha recebe) */
   381→  useEffect(() => {
   382→    const canvas = previewRef.current;
   383→    if (!canvas || !prepared) return;
   384→    const ctx = canvas.getContext("2d");
   385→    if (!ctx) return;
   386→    const cache = new Map<number, ImageData>();
   387→    let raf = 0;
   388→    let index = 0;
   389→    let nextAt = performance.now();
   390→    const draw = (now: number) => {
   391→      if (now >= nextAt) {
   392→        const frames = prepared.frames;
   393→        const frame = frames[index];
   394→        ctx.fillStyle = "#000000";
   395→        ctx.fillRect(0, 0, 128, 128);
   396→        if (showDeviceColors) {
   397→          let image = cache.get(index);
   398→          if (!image) {
   399→            image = decodeFrameToImageData(prepared.stream.data, index);
   400→            cache.set(index, image);
   401→          }
   402→          ctx.putImageData(image, 0, 0);
   403→        } else if (frame.bitmap) {
   404→          ctx.drawImage(frame.bitmap, 0, 0, 128, 128);
   405→        } else if (frame.image) {
   406→          ctx.putImageData(frame.image, 0, 0);
   407→        }
   408→        nextAt = now + Math.max(20, frame.delayMs);
   409→        index = (index + 1) % frames.length;
   410→      }
   411→      raf = requestAnimationFrame(draw);
   412→    };
   413→    raf = requestAnimationFrame(draw);
   414→    return () => cancelAnimationFrame(raf);
   415→  }, [prepared, showDeviceColors]);
   416→
   417→  const pct = progress.total > 0 ? Math.round((progress.sent / progress.total) * 100) : 0;
   418→
   419→  return (
   420→    <Card className="border-zinc-800 bg-zinc-900/60">
   421→      <CardHeader className="pb-3">
   422→        <CardTitle className="flex items-center gap-2 text-base">
   423→          <Monitor className="h-4 w-4 text-sky-400" /> Display, relógio & reset — cabo USB-C
   424→        </CardTitle>
   425→      </CardHeader>
   426→      <CardContent className="space-y-5">
   427→        <div className="grid gap-5 lg:grid-cols-[200px_1fr]">
   428→          <div className="space-y-2">
   429→            <div className="rounded-xl border-2 border-zinc-700 bg-zinc-950 p-1">
   430→              <canvas ref={previewRef} width={128} height={128} className="h-auto w-full rounded-lg" style={{ imageRendering: "pixelated" }} />
   431→            </div>
   432→            <p className="text-center text-[11px] text-zinc-500">
   433→              {prepared ? `prévia animada · ${prepared.frames.length} frames` : "prévia 128×128"}
   434→            </p>
   435→            {prepared && (
   436→              <Button variant="ghost" size="sm" onClick={() => setShowDeviceColors((v) => !v)} className="w-full text-[11px] text-zinc-400 hover:bg-zinc-800">
   437→                {showDeviceColors ? "Mostrando: cores da telinha (RGB565)" : "Mostrando: original — ver cores da telinha"}
   438→              </Button>
   439→            )}
   440→          </div>
   441→
   442→          <div className="space-y-4">
   443→            <div className="grid gap-3 sm:grid-cols-3">
   444→              <div className="space-y-1.5 sm:col-span-2">
   445→                <Label className="text-xs text-zinc-400">Imagem ou GIF (128×128 recomendado)</Label>
   446→                <Input
   447→                  type="file"
   448→                  accept="image/gif,image/png,image/jpeg,image/webp,image/bmp"
   449→                  onChange={(e) => onPrepare(e.target.files?.[0] ?? null)}
   450→                  disabled={uploading}
   451→                  className="border-zinc-700 bg-zinc-950 text-xs text-zinc-300 file:mr-3 file:rounded-md file:border-0 file:bg-zinc-800 file:px-3 file:py-1.5 file:text-xs file:text-zinc-200"
   452→                />
   453→              </div>
   454→              <div className="space-y-1.5">
   455→                <Label className="text-xs text-zinc-400">Slot (1–255)</Label>
   456→                <Input type="number" min={1} max={255} value={slot} onChange={(e) => setSlot(e.target.value)} disabled={uploading} className="border-zinc-700 bg-zinc-950 text-zinc-200" />
   457→              </div>
   458→            </div>
   459→
   460→            <div className="grid gap-3 sm:grid-cols-2">
   461→              <div className="space-y-1.5">
   462→                <Label className="text-xs text-zinc-400">Ajuste na tela</Label>
   463→                <Select value={fit} onValueChange={(v) => setFit(v as FitMode)} disabled={uploading}>
   464→                  <SelectTrigger className="border-zinc-700 bg-zinc-950 text-zinc-200">
   465→                    <SelectValue />
   466→                  </SelectTrigger>
   467→                  <SelectContent className="border-zinc-700 bg-zinc-950 text-zinc-200">
   468→                    <SelectItem value="contain">Fit (mantém proporção)</SelectItem>
   469→                    <SelectItem value="cover">Fill (corta)</SelectItem>
   470→                    <SelectItem value="stretch">Stretch (esticado)</SelectItem>
   471→                  </SelectContent>
   472→                </Select>
   473→              </div>
   474→              <div className="flex items-end gap-2">
   475→                <Button
   476→                  onClick={() => onGenerate("bounce")}
   477→                  disabled={uploading || busy !== null}
   478→                  variant="outline"
   479→                  className="flex-1 border-zinc-700 bg-transparent text-xs text-zinc-200 hover:bg-zinc-800"
   480→                >
   481→                  <Sparkles className="mr-1.5 h-3.5 w-3.5 text-cyan-300" /> Bola ciano
   482→                </Button>
   483→                <Button
   484→                  onClick={() => onGenerate("plasma")}
   485→                  disabled={uploading || busy !== null}
   486→                  variant="outline"
