    51→/* ------------------------------- copy block ------------------------------- */
    52→
    53→export function CopyBlock({ title, code }: { title?: string; code: string }) {
    54→  const [copied, setCopied] = useState(false);
    55→  const copy = async () => {
    56→    try {
    57→      await navigator.clipboard.writeText(code);
    58→      setCopied(true);
    59→      setTimeout(() => setCopied(false), 1600);
    60→    } catch {
    61→      f75log.err("Não foi possível copiar (clipboard bloqueado) — selecione o texto manualmente.");
    62→    }
    63→  };
    64→  return (
    65→    <div className="overflow-hidden rounded-lg border border-zinc-800 bg-zinc-950">
    66→      <div className="flex items-center justify-between border-b border-zinc-800 bg-zinc-900/80 px-3 py-1.5">
    67→        <span className="font-mono text-[11px] text-zinc-500">{title ?? "bash"}</span>
    68→        <Button variant="ghost" size="sm" onClick={copy} className="h-7 gap-1.5 px-2 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-emerald-400">
    69→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <ClipboardCopy className="h-3.5 w-3.5" />}
    70→          {copied ? "Copiado!" : "Copiar"}
    71→        </Button>
    72→      </div>
    73→      <pre className="scrollbar-thin max-h-72 overflow-auto p-3 font-mono text-xs leading-relaxed text-zinc-300">{code}</pre>
    74→    </div>
    75→  );
    76→}

   334→export function DisplayCard({
   335→  wiredConnected,
   336→  busy,
   337→  prepared,
   338→  onPrepare,
   339→  onGenerate,
   340→  onClear,
   341→  slot,
   342→  setSlot,
   343→  fit,
   344→  setFit,
   345→  progress,
   346→  uploading,
   347→  onCancelUpload,
   348→  onUpload,
   349→  onClock,
   350→  autoClock,
   351→  setAutoClock,
   352→  onFactoryReset,
   353→  resetArmed,
   354→}: {
   355→  wiredConnected: boolean;
   356→  busy: string | null;
   357→  prepared: PreparedUpload | null;
   358→  onPrepare: (file: File | null) => void;
   359→  onGenerate: (kind: "bounce" | "plasma") => void;
   360→  onClear: () => void;
   361→  slot: string;
   362→  setSlot: (v: string) => void;
   363→  fit: FitMode;
   364→  setFit: (f: FitMode) => void;
   365→  progress: { sent: number; total: number; eta: number };
   366→  uploading: boolean;
   367→  onCancelUpload: () => void;
   368→  onUpload: () => void;
   369→  onClock: () => void;
   370→  autoClock: boolean;
   371→  setAutoClock: (v: boolean) => void;
   372→  onFactoryReset: () => void;
   373→  resetArmed: boolean;
   374→}) {
   375→  const previewRef = useRef<HTMLCanvasElement | null>(null);
   376→  const [showDeviceColors, setShowDeviceColors] = useState(false);
   377→
   378→  /* animação da prévia (frames originais ou o RGB565 que a telinha recebe) */
   379→  useEffect(() => {
   380→    const canvas = previewRef.current;
   381→    if (!canvas || !prepared) return;
   382→    const ctx = canvas.getContext("2d");
   383→    if (!ctx) return;
   384→    const cache = new Map<number, ImageData>();
   385→    let raf = 0;
   386→    let index = 0;
   387→    let nextAt = performance.now();
   388→    const draw = (now: number) => {
   389→      if (now >= nextAt) {
   390→        const frames = prepared.frames;
   391→        const frame = frames[index];
   392→        ctx.fillStyle = "#000000";
   393→        ctx.fillRect(0, 0, 128, 128);
   394→        if (showDeviceColors) {
   395→          let image = cache.get(index);
   396→          if (!image) {
   397→            image = decodeFrameToImageData(prepared.stream.data, index);
   398→            cache.set(index, image);
   399→          }
   400→          ctx.putImageData(image, 0, 0);
   401→        } else if (frame.bitmap) {
   402→          ctx.drawImage(frame.bitmap, 0, 0, 128, 128);
   403→        } else if (frame.image) {
   404→          ctx.putImageData(frame.image, 0, 0);
   405→        }
   406→        nextAt = now + Math.max(20, frame.delayMs);
   407→        index = (index + 1) % frames.length;
   408→      }
   409→      raf = requestAnimationFrame(draw);
   410→    };
   411→    raf = requestAnimationFrame(draw);
   412→    return () => cancelAnimationFrame(raf);
   413→  }, [prepared, showDeviceColors]);
   414→
   415→  const pct = progress.total > 0 ? Math.round((progress.sent / progress.total) * 100) : 0;
   416→
   417→  return (
   418→    <Card className="border-zinc-800 bg-zinc-900/60">
   419→      <CardHeader className="pb-3">
   420→        <CardTitle className="flex items-center gap-2 text-base">
   421→          <Monitor className="h-4 w-4 text-sky-400" /> Display, relógio & reset — cabo USB-C
   422→        </CardTitle>
   423→      </CardHeader>
   424→      <CardContent className="space-y-5">
   425→        <div className="grid gap-5 lg:grid-cols-[200px_1fr]">
   426→          <div className="space-y-2">
   427→            <div className="rounded-xl border-2 border-zinc-700 bg-zinc-950 p-1">
   428→              <canvas ref={previewRef} width={128} height={128} className="h-auto w-full rounded-lg" style={{ imageRendering: "pixelated" }} />
   429→            </div>
   430→            <p className="text-center text-[11px] text-zinc-500">
   431→              {prepared ? `prévia animada · ${prepared.frames.length} frames` : "prévia 128×128"}
   432→            </p>
   433→            {prepared && (
   434→              <Button variant="ghost" size="sm" onClick={() => setShowDeviceColors((v) => !v)} className="w-full text-[11px] text-zinc-400 hover:bg-zinc-800">
   435→                {showDeviceColors ? "Mostrando: cores da telinha (RGB565)" : "Mostrando: original — ver cores da telinha"}
   436→              </Button>
   437→            )}
   438→          </div>
   439→
   440→          <div className="space-y-4">
   441→            <div className="grid gap-3 sm:grid-cols-3">
   442→              <div className="space-y-1.5 sm:col-span-2">
   443→                <Label className="text-xs text-zinc-400">Imagem ou GIF (128×128 recomendado)</Label>
   444→                <Input
   445→                  type="file"
   446→                  accept="image/gif,image/png,image/jpeg,image/webp,image/bmp"
   447→                  onChange={(e) => onPrepare(e.target.files?.[0] ?? null)}
   448→                  disabled={uploading}
   449→                  className="border-zinc-700 bg-zinc-950 text-xs text-zinc-300 file:mr-3 file:rounded-md file:border-0 file:bg-zinc-800 file:px-3 file:py-1.5 file:text-xs file:text-zinc-200"
   450→                />
   451→              </div>
   452→              <div className="space-y-1.5">
   453→                <Label className="text-xs text-zinc-400">Slot (1–255)</Label>
   454→                <Input type="number" min={1} max={255} value={slot} onChange={(e) => setSlot(e.target.value)} disabled={uploading} className="border-zinc-700 bg-zinc-950 text-zinc-200" />
   455→              </div>
   456→            </div>
   457→
   458→            <div className="grid gap-3 sm:grid-cols-2">
   459→              <div className="space-y-1.5">
   460→                <Label className="text-xs text-zinc-400">Ajuste na tela</Label>
   461→                <Select value={fit} onValueChange={(v) => setFit(v as FitMode)} disabled={uploading}>
   462→                  <SelectTrigger className="border-zinc-700 bg-zinc-950 text-zinc-200">
   463→                    <SelectValue />
   464→                  </SelectTrigger>
   465→                  <SelectContent className="border-zinc-700 bg-zinc-950 text-zinc-200">
   466→                    <SelectItem value="contain">Fit (mantém proporção)</SelectItem>
   467→                    <SelectItem value="cover">Fill (corta)</SelectItem>
   468→                    <SelectItem value="stretch">Stretch (esticado)</SelectItem>
   469→                  </SelectContent>
   470→                </Select>
   471→              </div>
   472→              <div className="flex items-end gap-2">
   473→                <Button
   474→                  onClick={() => onGenerate("bounce")}
   475→                  disabled={uploading || busy !== null}
   476→                  variant="outline"
   477→                  className="flex-1 border-zinc-700 bg-transparent text-xs text-zinc-200 hover:bg-zinc-800"
   478→                >
   479→                  <Sparkles className="mr-1.5 h-3.5 w-3.5 text-cyan-300" /> Bola ciano
   480→                </Button>
   481→                <Button
   482→                  onClick={() => onGenerate("plasma")}
   483→                  disabled={uploading || busy !== null}
   484→                  variant="outline"
   485→                  className="flex-1 border-zinc-700 bg-transparent text-xs text-zinc-200 hover:bg-zinc-800"

   566→interface KeyEventEntry {
   567→  id: number;
   568→  type: "down" | "up";
   569→  code: string;
   570→  key: string;
   571→  mods: string;
   572→}
   573→
   574→const INTERESTING = ["PrintScreen", "MetaLeft", "MetaRight", "AltLeft", "AltRight", "Tab", "F13", "F14", "F15", "Lang1", "Lang2"];
   575→
   576→export function KeyTesterCard() {
   577→  const [events, setEvents] = useState<KeyEventEntry[]>([]);
   578→  const seq = useRef(0);
   579→
   580→  useEffect(() => {
   581→    const push = (type: "down" | "up") => (e: KeyboardEvent) => {
   582→      if (e.key === "F12") return; // não atrapalhar o devtools
   583→      seq.current += 1;
   584→      const mods = [e.ctrlKey && "Ctrl", e.altKey && "Alt", e.metaKey && "Win/Super", e.shiftKey && "Shift"].filter(Boolean).join("+");
   585→      setEvents((prev) => [...prev.slice(-24), { id: seq.current, type, code: e.code, key: e.key, mods }]);
   586→    };
   587→    const down = push("down");
   588→    const up = push("up");
   589→    window.addEventListener("keydown", down);
   590→    window.addEventListener("keyup", up);
   591→    return () => {
   592→      window.removeEventListener("keydown", down);
   593→      window.removeEventListener("keyup", up);
   594→    };
   595→  }, []);
   596→
   597→  return (
   598→    <Card className="border-zinc-800 bg-zinc-900/60">
   599→      <CardHeader className="pb-3">
   600→        <CardTitle className="flex items-center gap-2 text-base">

   602→        </CardTitle>
   603→      </CardHeader>
   604→      <CardContent className="space-y-3">
   605→        <p className="text-xs leading-relaxed text-zinc-400">
   606→          Aperte teclas aqui pra ver o <code className="rounded bg-zinc-800 px-1">event.code</code> real que o CachyOS recebe.{" "}
   607→          <strong className="text-zinc-200">Alt+Tab saudável = o seletor de janelas abre e a página perde foco</strong> (o GNOME consome o atalho
   608→          no compositor). Se apertar Alt+Tab <em>não acontecer nada</em>, é o firmware do teclado travando — resolva no card{" "}
   609→          <em>Desempenho &amp; Modo Jogo</em>. Use também pra descobrir o código da tecla Print (captura de tela) e rebindar no card Sistema.
   610→        </p>
   611→        <div className="scrollbar-thin max-h-44 space-y-1 overflow-y-auto rounded-lg border border-zinc-800 bg-zinc-950 p-3 font-mono text-xs">

   614→          ) : (
   615→            events.map((ev) => (
   616→              <p key={ev.id} className={INTERESTING.includes(ev.code) ? "text-emerald-300" : "text-zinc-400"}>
   617→                <span className={ev.type === "down" ? "text-sky-400" : "text-zinc-600"}>{ev.type === "down" ? "▼" : "△"}</span>{" "}
   618→                <span className="text-zinc-200">{ev.code}</span>
   619→                {ev.mods && <span className="text-amber-400/90"> [{ev.mods}]</span>}
   620→                <span className="text-zinc-600"> (key &quot;{ev.key}&quot;)</span>
   621→              </p>
   622→            ))
   623→          )}
   624→        </div>
   625→        <div className="flex justify-end">
   626→          <Button variant="ghost" size="sm" onClick={() => setEvents([])} className="h-7 gap-1.5 px-2 text-xs text-zinc-500 hover:bg-zinc-800">
   627→            <Eraser className="h-3.5 w-3.5" /> limpar
   628→          </Button>
   629→        </div>
   630→      </CardContent>
   631→    </Card>
   632→  );
